      || list.closest('[data-testid="virtuoso-scroller"]')
      || list.parentElement;

  function hideSignalCard(card) {
    const item = card?.closest?.(".virtuoso-grid-item") || card;
    if (!item) return;
    item.classList.add(HIDDEN_CLASS);
    item.dataset.spreadradarHidden = "true";
  }

  function addCloseButton(card) {
    if (!card || card.dataset.spreadradarClose === "true") return;

    card.dataset.spreadradarClose = "true";
    if (getComputedStyle(card).position === "static") {
      card.style.position = "relative";
    }

    const close = document.createElement("button");
    close.type = "button";
    close.className = CLOSE_CLASS;
    close.textContent = "×";
    close.title = "Hide signal";

    const stopCloseEvent = (event) => {
      event.preventDefault();
      event.stopPropagation();
      event.stopImmediatePropagation?.();
    };

    close.addEventListener("pointerdown", stopCloseEvent, true);
    close.addEventListener("mousedown", stopCloseEvent, true);
    close.addEventListener("click", (event) => {
      stopCloseEvent(event);
      hideSignalCard(card);
    }, true);

    card.appendChild(close);
  }

  function clearOldRoots(currentRoot) {
    $$(`.${PANEL_ROOT_CLASS}`).forEach((root) => {
      if (root !== currentRoot) root.classList.remove(PANEL_ROOT_CLASS);
    });
  }

  function syncSignalPanel() {
    injectStyle();

    const shouldDisable = isUtilityPage();
    const list = shouldDisable ? null : findSignalsList();
    const root = list ? findPanelRoot(list) : null;
    const enabled = Boolean(root && window.innerWidth > 1200);

    clearOldRoots(root);

    document.body.classList.toggle(PANEL_ON_CLASS, enabled);
    document.body.classList.toggle(SIGNALS_RESERVED_CLASS, enabled);

    if (!enabled) {
      $$(`.${CLOSE_CLASS}`).forEach((button) => button.remove());
      if (root) root.classList.remove(PANEL_ROOT_CLASS);
      return;
    }

    root.classList.add(PANEL_ROOT_CLASS);
    $$(".signal-card", list).forEach(addCloseButton);
  }

  function start() {
    injectStyle();
    syncSignalPanel();
    setInterval(syncSignalPanel, 1000);
    window.addEventListener("resize", syncSignalPanel, { passive: true });

    const observer = new MutationObserver(() => syncSignalPanel());
    observer.observe(document.documentElement, { childList: true, subtree: true });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", start, { once: true });
  } else {
    start();
  }
