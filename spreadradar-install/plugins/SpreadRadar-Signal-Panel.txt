  function updateScrollbarWidthVar() {
    try {
      const width = Math.max(0, window.innerWidth - document.documentElement.clientWidth);
      document.documentElement.style.setProperty(SCROLLBAR_WIDTH_VAR, `${width || 17}px`);
    } catch (_) {}
  }

  function isElementVisible(el) {
    if (!(el instanceof Element)) return false;
    const style = getComputedStyle(el);
    if (style.display === "none" || style.visibility === "hidden") return false;
    const rect = el.getBoundingClientRect();
    return rect.width > 0 && rect.height > 0;
  }

  function setChartFsSuspended(value) {
    chartFsSuspended = Boolean(value);
    document.body?.classList?.toggle(CHART_FULLSCREEN_CLASS, chartFsSuspended);
  }

  function getChartFsButtonMode(button) {
    const text = String(button?.textContent || "").trim();
    if (text === "⛶") return "open";
    if (text === "✕" || text === "×" || text.toLowerCase() === "x") return "close";
    return "unknown";
  }

  function handleChartFsButtonClick(button) {
    const mode = getChartFsButtonMode(button);

    if (mode === "open") {
      setChartFsSuspended(true);
      syncSignalPanel();
      setTimeout(syncSignalPanel, 80);
      setTimeout(syncSignalPanel, 250);
      return;
    }

    if (mode === "close") {
      setTimeout(() => {
        setChartFsSuspended(false);
        syncSignalPanel();
      }, 120);
      setTimeout(syncSignalPanel, 260);
    }
  }

  function normalizeKeyText(value) {
    return String(value || "")
      .replace(/\s+/g, " ")
      .replace(/[×✕]/g, "")
      .trim()
      .toLowerCase();
  }

  function normalizeHiddenEntry(value) {
    const now = Date.now();
    if (typeof value === "number") {
      return { hiddenAt: value, lastSeenAt: value, absentSince: 0 };
    }
    if (value && typeof value === "object") {
      return {
        hiddenAt: Number(value.hiddenAt || now),
        lastSeenAt: Number(value.lastSeenAt || value.hiddenAt || now),
        absentSince: Number(value.absentSince || 0)
      };
    }
    return null;
  }

  function loadHiddenSignals() {
    try {
      const parsed = JSON.parse(sessionStorage.getItem(HIDDEN_SIGNALS_KEY) || "{}");
      if (!parsed || typeof parsed !== "object") return {};
      const normalized = {};
      Object.entries(parsed).forEach(([key, value]) => {
        const entry = normalizeHiddenEntry(value);
        if (entry) normalized[key] = entry;
      });
      return normalized;
    } catch (_) {
      return {};
    }
  }

  function saveHiddenSignals(hidden) {
    try {
      sessionStorage.setItem(HIDDEN_SIGNALS_KEY, JSON.stringify(hidden || {}));
    } catch (_) {}
  }

  function getCurrentSignalCardKeys() {
    const keys = new Set();
    document.querySelectorAll(`${SIGNAL_LIST_SELECTOR} .signal-card`).forEach(card => {
      const key = getSignalCardKey(card);
      if (key) keys.add(key);
    });
    return keys;
  }

  function cleanupHiddenSignals(hidden = loadHiddenSignals(), presentKeys = getCurrentSignalCardKeys()) {
    const now = Date.now();
    let changed = false;

    Object.entries(hidden).forEach(([key, value]) => {
      const entry = normalizeHiddenEntry(value);
      if (!entry) {
        delete hidden[key];
        changed = true;
        return;
      }

      if (presentKeys.has(key)) {
        if (entry.absentSince) changed = true;
        entry.absentSince = 0;
        entry.lastSeenAt = now;
        hidden[key] = entry;
        return;
      }

      if (!entry.absentSince) {
        entry.absentSince = now;
        hidden[key] = entry;
        changed = true;
        return;
      }

      if (now - entry.absentSince >= HIDDEN_SIGNAL_ABSENT_CLEAR_MS) {
        delete hidden[key];
        changed = true;
        return;
      }

      hidden[key] = entry;
    });

    if (changed) saveHiddenSignals(hidden);
    return hidden;
  }

  function getSignalCardKey(card) {
    if (!(card instanceof Element)) return "";

    const explicit = card.getAttribute("data-route-key") || card.getAttribute("data-signal-key") || card.dataset?.routeKey || card.dataset?.signalKey;
    if (explicit) return `explicit:${normalizeKeyText(explicit)}`;

    const symbol = normalizeKeyText(
      card.querySelector?.("td.symbol, .symbol, [data-help='signal.symbol']")?.textContent ||
      card.querySelector?.(".header td:first-child")?.textContent ||
      ""
    );

    const longExchange = normalizeKeyText(card.querySelector?.("table.exchange-info td.long, table.exchange-info .long, .exchange-info .long")?.textContent || "");
    const shortExchange = normalizeKeyText(card.querySelector?.("table.exchange-info td.short, table.exchange-info .short, .exchange-info .short")?.textContent || "");

    if (symbol && (longExchange || shortExchange)) {
      return `route:${symbol}|long:${longExchange}|short:${shortExchange}`;
    }

    const headerText = normalizeKeyText(card.querySelector?.(".header")?.textContent || "");
    const exchangesText = normalizeKeyText(card.querySelector?.(".exchange-info")?.textContent || "");
    const footerText = normalizeKeyText(card.querySelector?.(".signal-footer")?.textContent || "");
    const compact = `${headerText}|${exchangesText}|${footerText}`.slice(0, 360);
    return compact ? `text:${compact}` : "";
  }

  function rememberHiddenCard(card) {
    const key = getSignalCardKey(card);
    if (!key) return;
    const hidden = cleanupHiddenSignals();
    const now = Date.now();
    hidden[key] = { hiddenAt: now, lastSeenAt: now, missingScans: 0 };
    saveHiddenSignals(hidden);
  }

  function isSignalCardRememberedHidden(card, hidden = cleanupHiddenSignals()) {
    const key = getSignalCardKey(card);
    return Boolean(key && hidden[key]);
  }

  function forceHideSignalCard(card) {
    if (!(card instanceof Element)) return;
    const item = card.closest?.(".virtuoso-grid-item") || card.parentElement;

    card.dataset.spreadradarHidden = "true";
    card.style.setProperty("display", "none", "important");

    if (item instanceof Element && item !== card) {
      item.dataset.spreadradarHidden = "true";
      item.style.setProperty("display", "none", "important");
    }
  }

  function applyRememberedHiddenSignals() {
    const presentKeys = getCurrentSignalCardKeys();
    const hidden = cleanupHiddenSignals(loadHiddenSignals(), presentKeys);
    document.querySelectorAll(`${SIGNAL_LIST_SELECTOR} .signal-card`).forEach(card => {
      if (isSignalCardRememberedHidden(card, hidden)) {
        forceHideSignalCard(card);
      }
    });
  }


  function hideSignalCard(card) {
    if (!card) return;
    rememberHiddenCard(card);
    forceHideSignalCard(card);
    requestAnimationFrame(syncSignalPanel);
  }

  function stopCloseEvent(event) {
    event.preventDefault();
    event.stopPropagation();
    event.stopImmediatePropagation?.();
  }

  function addSignalCloseButtons() {
    applyRememberedHiddenSignals();
    document.querySelectorAll(`${SIGNAL_LIST_SELECTOR} .signal-card:not([data-spreadradar-close])`).forEach(card => {
      if (card.dataset.spreadradarHidden === "true") return;
      card.dataset.spreadradarClose = "true";
      if (getComputedStyle(card).position === "static") {
        card.style.setProperty("position", "relative", "important");
      }

      const close = document.createElement("button");
      close.type = "button";
      close.className = "spreadradar-signal-close";
      close.textContent = "×";
      close.title = "Hide signal";
      close.setAttribute("aria-label", "Hide signal");

      const closeNow = event => {
        stopCloseEvent(event);
        hideSignalCard(card);
      };

      close.addEventListener("pointerdown", closeNow, true);
      close.addEventListener("mousedown", stopCloseEvent, true);
      close.addEventListener("touchstart", closeNow, true);
      close.addEventListener("click", closeNow, true);

      card.appendChild(close);
    });
  }

  function hasVisibleSignalCards() {
    const list = document.querySelector(SIGNAL_LIST_SELECTOR);
    if (!list || !list.isConnected) return false;

    return Array.from(list.querySelectorAll(".signal-card")).some(card => {
      if (!(card instanceof Element)) return false;
      if (card.dataset.spreadradarHidden === "true") return false;
      const style = getComputedStyle(card);
      if (style.display === "none" || style.visibility === "hidden") return false;
      const rect = card.getBoundingClientRect();
      return rect.width > 0 && rect.height > 0;
    });
  }

  function setPanelEnabled(enabled) {
    if (lastPanelState === enabled && document.body.classList.contains(BODY_CLASS) === enabled) return;
    lastPanelState = enabled;
    document.body.classList.toggle(BODY_CLASS, enabled);
  }

  function syncSignalPanel() {
    updateScrollbarWidthVar();
    document.body.classList.toggle(CHART_FULLSCREEN_CLASS, chartFsSuspended);
    const enabled = !chartFsSuspended && window.innerWidth > DESKTOP_MIN_WIDTH && hasVisibleSignalCards();
    setPanelEnabled(enabled);

    if (!enabled) {
      document.querySelectorAll(".spreadradar-signal-close").forEach(btn => btn.remove());
      document.querySelectorAll(`${SIGNAL_LIST_SELECTOR} .signal-card[data-spreadradar-close]`).forEach(card => {
        delete card.dataset.spreadradarClose;
      });
      return;
    }

    addSignalCloseButtons();
  }

  function watchDom() {
    if (observer) observer.disconnect();
    observer = new MutationObserver(() => {
      applyRememberedHiddenSignals();
      syncSignalPanel();
    });
    observer.observe(document.documentElement || document.body, {
      childList: true,
      subtree: true
    });
  }

  function watchSpaNavigation() {
    const checkLocation = () => {
      if (location.pathname !== lastPath || location.search !== lastSearch) {
        lastPath = location.pathname;
        lastSearch = location.search;
        syncSignalPanel();
      }
    };

    const wrapHistoryMethod = (methodName) => {
      const original = history[methodName];
      if (typeof original !== "function" || original.__spreadradarPanelWrapped) return;
      const wrapped = function (...args) {
        const result = original.apply(this, args);
        setTimeout(checkLocation, 0);
        return result;
      };
      wrapped.__spreadradarPanelWrapped = true;
      history[methodName] = wrapped;
    };

    wrapHistoryMethod("pushState");
    wrapHistoryMethod("replaceState");
    window.addEventListener("popstate", checkLocation, true);
  }

  function init() {
    updateScrollbarWidthVar();
    injectStyle();
    watchDom();
    watchSpaNavigation();
    applyRememberedHiddenSignals();
    syncSignalPanel();
    window.addEventListener("resize", syncSignalPanel, { passive: true });
    document.addEventListener("click", event => {
      const button = event.target?.closest?.("button.chart-fs-btn, .chart-fs-btn");
      if (button) handleChartFsButtonClick(button);
    }, true);
    setInterval(syncSignalPanel, SYNC_INTERVAL_MS);
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", init, { once: true });
  } else {
    init();
  }
})();
