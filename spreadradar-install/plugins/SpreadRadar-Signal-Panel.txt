// ==UserScript==
// @name         Signal Panel Right Layout for AT
// @namespace    https://www.arbitterminal.online/
// @version      1.08-virtuoso-safe
// @description  Окремий модуль стилізації signal panel, utility-page логіки і хрестика закриття signal-card, сумісний зі SpreadRadar Links
// @match        https://www.arbitterminal.online/*
// @match        https://arbitterminal.online/*
// @grant        none
// @connect      arbitterminal.github.io
// @updateURL    https://arbitterminal.github.io/spreadradar-install/plugins/SpreadRadar-Signal-Panel.txt
// @downloadURL  https://arbitterminal.github.io/spreadradar-install/plugins/SpreadRadar-Signal-Panel.txt
// @run-at       document-idle
// ==/UserScript==

(function () {
  "use strict";

  if (!/(^|\.)arbitterminal\.online$/i.test(location.hostname)) return;

  const VERSION = "1.08-virtuoso-safe";
  const STYLE_ID = "spreadradar-signal-panel-style-v108";
  const PANEL_ROOT_CLASS = "spreadradar-signal-panel-root-v108";
  const PANEL_ON_CLASS = "spreadradar-signals-panel-on-v108";
  const CLOSE_CLASS = "spreadradar-signal-close-v108";
  const HIDDEN_CLASS = "spreadradar-signal-item-hidden-v108";

  let syncTimer = 0;
  let lastRoot = null;

  const $ = (selector, root = document) => root.querySelector(selector);
  const $$ = (selector, root = document) => Array.from(root.querySelectorAll(selector));

  function isUtilityPage() {
    const path = location.pathname.toLowerCase();
    return /history|trend|chart|leaderboard|cooperation|settings|login/.test(path)
      || Boolean(document.querySelector(".chart-fullscreen-btn:fullscreen"));
  }

  function isVisible(el) {
    if (!(el instanceof Element) || !el.isConnected) return false;
    const rect = el.getBoundingClientRect();
    const style = getComputedStyle(el);
    return rect.width > 0 && rect.height > 0 && style.display !== "none" && style.visibility !== "hidden";
  }

  function findSignalsList() {
    return $(".signals-list[data-testid='virtuoso-item-list'], .signals-list");
  }

  function findPanelRoot(list) {
    if (!list) return null;

    // New AT layout: React Virtuoso.
    // Important: do NOT position/flex/transform .signals-list itself.
    // Use the Virtuoso scroller as the panel root and leave list/items in React's expected flow.
    return list.closest('[data-virtuoso-scroller="true"]')
      || list.closest('[data-testid="virtuoso-scroller"]')
      || list.parentElement;
  }

  function injectStyle() {
    let style = document.getElementById(STYLE_ID);
    if (!style) {
      style = document.createElement("style");
      style.id = STYLE_ID;
      document.head.appendChild(style);
    }

    style.textContent = `
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS}{
        position:fixed!important;
        left:auto!important;
        right:6px!important;
        top:49px!important;
        bottom:22px!important;
        width:450px!important;
        height:calc(100vh - 71px)!important;
        max-height:calc(100vh - 71px)!important;
        min-height:0!important;
        overflow-y:auto!important;
        overflow-x:hidden!important;
        z-index:100000!important;
        background:transparent!important;
        box-sizing:border-box!important;
        padding:0!important;
        margin:0!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signals-list{
        width:100%!important;
        min-width:0!important;
        max-width:none!important;
        box-sizing:border-box!important;
        padding-left:6px!important;
        padding-right:8px!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .virtuoso-grid-item.${HIDDEN_CLASS}{
        display:none!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card{
        width:100%!important;
        max-width:none!important;
        box-sizing:border-box!important;
        margin:0 0 8px 0!important;
        padding:8px 10px 10px!important;
        border-radius:6px!important;
        position:relative!important;
        overflow:visible!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card table{
        width:100%!important;
        table-layout:auto!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card td,
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card th{
        min-width:0!important;
        overflow:hidden!important;
        text-overflow:ellipsis!important;
        white-space:nowrap!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .header td:first-child,
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card td.symbol,
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card [data-help="signal.symbol"]{
        width:82px!important;
        max-width:82px!important;
        font-weight:800!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .header-right,
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card [data-help="signal.spreadValue"]{
        text-align:right!important;
        padding-right:28px!important;
        overflow:visible!important;
        text-overflow:clip!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .exchange-info td:first-child{
        width:118px!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .exchange-info td:nth-child(2),
      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .exchange-info td:nth-child(3){
        width:auto!important;
        overflow:visible!important;
        text-overflow:clip!important;
        white-space:nowrap!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .signal-footer{
        display:grid!important;
        grid-template-columns:112px 112px minmax(92px,1fr)!important;
        align-items:center!important;
        gap:8px!important;
        min-height:38px!important;
        height:auto!important;
        overflow:visible!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .signal-footer button{
        width:112px!important;
        min-width:112px!important;
        max-width:112px!important;
        height:31px!important;
        min-height:31px!important;
        max-height:31px!important;
        padding:0 10px!important;
        border-radius:5px!important;
        font:700 13px/1 Arial,sans-serif!important;
        display:inline-flex!important;
        align-items:center!important;
        justify-content:center!important;
        white-space:nowrap!important;
        overflow:hidden!important;
        text-overflow:ellipsis!important;
        box-sizing:border-box!important;
        margin:0!important;
        background:linear-gradient(#ffffff,#f1f5f9)!important;
        color:#111827!important;
        border:1px solid #aeb8c6!important;
        box-shadow:0 1px 2px rgba(15,23,42,.12)!important;
        cursor:pointer!important;
      }

      body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS} .signal-card .signal-footer .signal-time{
        display:block!important;
        min-width:0!important;
        overflow:visible!important;
        text-overflow:clip!important;
        white-space:nowrap!important;
        font-size:12px!important;
        line-height:1.25!important;
        font-weight:800!important;
        text-align:left!important;
      }

      .${CLOSE_CLASS}{
        position:absolute!important;
        top:3px!important;
        right:5px!important;
        left:auto!important;
        width:20px!important;
        height:20px!important;
        padding:0!important;
        margin:0!important;
        box-sizing:border-box!important;
        border:0!important;
        border-radius:5px!important;
        background:#dc3545!important;
        color:#fff!important;
        font:700 14px/20px Arial,sans-serif!important;
        display:flex!important;
        align-items:center!important;
        justify-content:center!important;
        cursor:pointer!important;

        opacity:.85!important;

      }

      .${CLOSE_CLASS}:hover{
        opacity:1!important;
      }

      @media(max-width:1200px){
        body.${PANEL_ON_CLASS} .${PANEL_ROOT_CLASS}{
          position:static!important;
          width:auto!important;
          height:auto!important;
          max-height:none!important;
          overflow:visible!important;
        }
      }
    `;
  }

  function hideSignalCard(card) {
    const item = card?.closest?.(".virtuoso-grid-item") || card;
    if (!item) return;
    item.classList.add(HIDDEN_CLASS);
    item.dataset.spreadradarHidden = "true";
  }

  function addCloseButton(card) {
    if (!card || card.dataset.spreadradarCloseV108 === "true") return;

    card.dataset.spreadradarCloseV108 = "true";
    if (getComputedStyle(card).position === "static") card.style.position = "relative";

    const close = document.createElement("button");
    close.type = "button";
    close.className = CLOSE_CLASS;
    close.textContent = "×";
    close.title = "Hide signal";

    const stopCloseEvent = (event) => {
      event.preventDefault();
      event.stopPropagation();
      event.stopImmediatePropagation?.();
    };

    close.addEventListener("pointerdown", stopCloseEvent, true);
    close.addEventListener("mousedown", stopCloseEvent, true);
    close.addEventListener("click", (event) => {
      stopCloseEvent(event);
      hideSignalCard(card);
    }, true);

    card.appendChild(close);
  }

  function removeOldV107Classes() {
    document.body.classList.remove("spreadradar-signals-panel-on", "spreadradar-signals-reserved");
    $$(".spreadradar-signal-panel-root").forEach((root) => root.classList.remove("spreadradar-signal-panel-root"));
  }

  function clearOldRoots(currentRoot) {
    $$(`.${PANEL_ROOT_CLASS}`).forEach((root) => {
      if (root !== currentRoot) root.classList.remove(PANEL_ROOT_CLASS);
    });
  }

  function syncSignalPanel() {
    injectStyle();
    removeOldV107Classes();

    const shouldDisable = isUtilityPage();
    const list = shouldDisable ? null : findSignalsList();
    const root = list ? findPanelRoot(list) : null;
    const enabled = Boolean(root && isVisible(root) && window.innerWidth > 1200);

    clearOldRoots(root);
    document.body.classList.toggle(PANEL_ON_CLASS, enabled);

    if (!enabled) {
      if (lastRoot) lastRoot.classList.remove(PANEL_ROOT_CLASS);
      lastRoot = null;
      return;
    }

    root.classList.add(PANEL_ROOT_CLASS);
    lastRoot = root;
    $$(".signal-card", list).forEach(addCloseButton);
  }

  function scheduleSync(delay = 80) {
    if (syncTimer) return;
    syncTimer = window.setTimeout(() => {
      syncTimer = 0;
      syncSignalPanel();
    }, delay);
  }

  function start() {
    injectStyle();
    syncSignalPanel();
    window.addEventListener("resize", () => scheduleSync(120), { passive: true });
    window.addEventListener("hashchange", () => scheduleSync(120), { passive: true });

    const observer = new MutationObserver(() => scheduleSync(120));
    observer.observe(document.body || document.documentElement, { childList: true, subtree: true });

    // Fallback only, not an aggressive layout loop.
    setInterval(() => scheduleSync(0), 2500);
    console.log(`[Signal Panel Right Layout for AT] ${VERSION} loaded`);
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", start, { once: true });
  } else {
    start();
  }
})();
