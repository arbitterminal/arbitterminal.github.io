// Early native FastTrade URL override for the new ArbitTerminal format:
// ?s=AERGO|binance|bingx|L&ft=1&sp=0.32
(function mixEarlyNativeFastTradeOpenOverride() {
  "use strict";

  const SPREADRADAR_SETTINGS_KEY = "mix_spreadradar_settings_v1";

  function formatSpreadValue(value) {
    const n = Number(String(value ?? "").trim().replace(",", "."));
    if (!Number.isFinite(n)) return "";
    return String(Math.round(n * 1000) / 1000);
  }

  try {
    const params = new URLSearchParams(window.location.search);
    if (params.get("ft") !== "1" || !params.has("s")) return;

    const settings = JSON.parse(localStorage.getItem(SPREADRADAR_SETTINGS_KEY) || "{}");
    if (settings?.fastTradeOpenOverrideEnabled !== true) return;

    const customSpread = formatSpreadValue(settings.fastTradeOpenSpread);
    if (!customSpread) return;
    if (params.get("sp") === customSpread) return;

    params.set("sp", customSpread);
    const query = params.toString();
    const nextUrl = `${window.location.pathname}${query ? `?${query}` : ""}${window.location.hash}`;
    window.history.replaceState(window.history.state, document.title, nextUrl);
  } catch (err) {
    console.warn("[MIX] Native FastTrade Open override failed:", err);
  }
})();

      const n = normalizeSoundsVariant(primary);
      if (n !== primary) writeLS(SELECTOR_STORAGE_KEY_SOUNDS.key, n);
      return n;
    }
    for (const legacyKey of SELECTOR_STORAGE_KEY_SOUNDS.legacyKeys || []) {
      const raw = readLS(legacyKey);
      if (raw == null) continue;
      const n = normalizeSoundsVariant(raw);
      writeLS(SELECTOR_STORAGE_KEY_SOUNDS.key, n);
      return n;
    }
    return "v1";
  }

  function getSoundsVariantIds() {
    const api = window.__AT_OC_SOUNDS__;
    if (typeof api?.listProfiles === "function") {
      try {
        const ids = api.listProfiles();
        const supported = SOUNDS_SELECTOR_VARIANT_IDS.filter(id => Array.isArray(ids) && ids.includes(id));
        if (supported.length) return ["off", ...supported];
      } catch (_) {}
    }
    return ["off", ...SOUNDS_SELECTOR_VARIANT_IDS];
  }

  function getSoundsVariant() {
    const api = window.__AT_OC_SOUNDS__;
    if (typeof api?.getProfileId === "function") {
      try { return normalizeSoundsVariant(api.getProfileId()); } catch (_) {}
    }
    return readSoundsVariantStorage();
  }

  function setSoundsVariant(id, { syncPanel = true, syncStatus = true } = {}) {
    const next = normalizeSoundsVariant(id);
    const api = window.__AT_OC_SOUNDS__;
    if (typeof api?.setProfile === "function") {
      try { api.setProfile(next); } catch (_) {}
    } else {
      writeLS(SELECTOR_STORAGE_KEY_SOUNDS.key, next);
    }
    if (syncPanel) renderPanelState();
    if (syncStatus) setStatus(tr().saved);
  }

  function getSoundsSelectorValue() {
    return state.sounds === false ? "off" : getSoundsVariant();
  }

  function setSoundsSelectorValue(id) {
    const next = normalizeSoundsMode(id);
    if (next === "off") {
      window.__MIX_MODULES__.setEnabled("sounds", false);
      setStatus(tr().saved);
      return;
    }
    setSoundsVariant(next, { syncPanel: false, syncStatus: false });
    window.__MIX_MODULES__.setEnabled("sounds", true);
    renderPanelState();
    setStatus(tr().saved);
  }

  function tr() { return I18N[lang] || I18N.en; }

  function getModule(id) { return MODULES.find(m => m.id === id) || null; }

  function hasPendingReloadChanges() {
    return MODULES.some(m => m.apply === "reload" && draftState[m.id] !== state[m.id]);
  }

  function setStatus(text) {
    const panel = document.getElementById(PANEL_ID);
    if (!panel) return;
    const status = panel.querySelector(".mix-modules-status");
    if (!status) return;
    status.textContent = text || "";
    if (statusTimer) clearTimeout(statusTimer);
    if (text) {
      statusTimer = setTimeout(() => {
        status.textContent = hasPendingReloadChanges() ? tr().pendingReload : "";
      }, 1600);
    }
  }

  function setSelectorButtonState(btn, active) {
    btn.classList.toggle("is-active", active);
    btn.setAttribute("aria-pressed", active ? "true" : "false");
  }

  function syncSelectorButtons(panel, selector, getCurrentValue, getButtonValue) {
    const currentValue = getCurrentValue();
    panel.querySelectorAll(selector).forEach(btn => {
      if (btn instanceof HTMLButtonElement) {
        setSelectorButtonState(btn, getButtonValue(btn) === currentValue);
      }
    });
  }

  function renderPanelState() {
    const panel = document.getElementById(PANEL_ID);
    if (!panel) return;

    MODULES.forEach(mod => {
      const input = panel.querySelector(`input[data-mix-module="${mod.id}"]`);
      if (input) input.checked = draftState[mod.id] !== false;
    });

    panel.querySelector(".mix-modules-apply")?.textContent && (panel.querySelector(".mix-modules-apply").textContent = tr().apply);
    panel.querySelector(".mix-modules-hint") && (panel.querySelector(".mix-modules-hint").textContent = tr().reloadHint);

    syncSelectorButtons(panel, "button[data-mix-sounds-variant]", getSoundsSelectorValue, btn => btn.dataset.mixSoundsVariant);

    const pnl = getPnlSettings();
    const netPnlInput = panel.querySelector('input[data-mix-pnl-setting="netPnlEnabled"]');
    const pnlLevelSelect = panel.querySelector('select[data-mix-pnl-setting="pnlLevel"]');
    const cashbackInput = panel.querySelector('input[data-mix-pnl-setting="feeCashbackEnabled"]');
    const cashbackPercent = panel.querySelector('input[data-mix-pnl-setting="feeCashbackPercent"]');
    if (netPnlInput) netPnlInput.checked = pnl.netPnlEnabled;
    if (pnlLevelSelect) pnlLevelSelect.value = String(pnl.pnlLevel);
    if (cashbackInput) cashbackInput.checked = pnl.feeCashbackEnabled;
    if (cashbackPercent) cashbackPercent.value = String(pnl.feeCashbackPercent);

    const spreadRadar = getSpreadRadarSettings();
    const spreadRadarValue = panel.querySelector("[data-mix-spreadradar-order-value]");
    const spreadRadarRange = panel.querySelector('input[data-mix-spreadradar-setting="orderUsdt"]');
    const fastTradeOpenOverride = panel.querySelector('input[data-mix-spreadradar-setting="fastTradeOpenOverrideEnabled"]');
    const fastTradeOpenSpread = panel.querySelector('input[data-mix-spreadradar-setting="fastTradeOpenSpread"]');
    if (spreadRadarValue) spreadRadarValue.textContent = `$${spreadRadar.orderUsdt}`;
    if (spreadRadarRange) spreadRadarRange.value = String(spreadRadar.orderUsdt);
    if (fastTradeOpenOverride) fastTradeOpenOverride.checked = spreadRadar.fastTradeOpenOverrideEnabled === true;
    if (fastTradeOpenSpread) {
      fastTradeOpenSpread.value = String(spreadRadar.fastTradeOpenSpread);
      fastTradeOpenSpread.disabled = spreadRadar.fastTradeOpenOverrideEnabled !== true;
    }
    syncSelectorButtons(panel, "button[data-mix-spreadradar-order]", () => String(spreadRadar.orderUsdt), btn => btn.dataset.mixSpreadradarOrder);

    const status = panel.querySelector(".mix-modules-status");
    if (status) status.textContent = hasPendingReloadChanges() ? tr().pendingReload : "";
  }

  function notify(id, enabled) {
    (listeners[id] || []).forEach(fn => {
      try { fn(enabled); } catch (e) { console.warn("MIX toggle listener failed:", id, e); }
    });
  }

  /* ---------- публічний API ---------- */

  window.__MIX_MODULES__ = {
    isEnabled(id) { return state[id] !== false; },

    setEnabled(id, enabled) {
      const mod = getModule(id);
      if (!mod) return;
      const next = enabled !== false;
      const stateChanged = state[id] !== next;
      const draftChanged = draftState[id] !== next;
      if (!stateChanged && !draftChanged) return;
      state[id] = next;
      draftState[id] = next;
      saveState();
      renderPanelState();
      if (stateChanged) notify(id, next);
    },

    setDraftEnabled(id, enabled) {
      const mod = getModule(id);
      if (!mod) return;
      const next = enabled !== false;
      if (mod.apply === "reload") {
        if (draftState[id] === next) return;
        draftState[id] = next;
        renderPanelState();
        return;
      }
      this.setEnabled(id, next);
    },

    applyChanges() {
      const needsReload = hasPendingReloadChanges();
      MODULES.forEach(mod => {
        if (mod.apply === "reload") state[mod.id] = draftState[mod.id] !== false;
      });
      saveState();
      renderPanelState();
      if (needsReload) { location.reload(); return; }
      setStatus(tr().saved);
    },

    onToggle(id, fn) {
      if (typeof fn !== "function") return;
      if (!listeners[id]) listeners[id] = [];
      listeners[id].push(fn);
    },

    selectors: {
      soundsVariant: {
        list: getSoundsVariantIds,
        get: getSoundsSelectorValue,
        set: setSoundsSelectorValue,
        normalize: normalizeSoundsMode
      }
    },

    pnlSettings: {
      get: getPnlSettings,
      set: setPnlSettings,
      levels: Object.assign({}, PNL_LEVELS)
    },

    spreadRadarSettings: {
      get: getSpreadRadarSettings,
      set: setSpreadRadarSettings,
      orderSizes: SPREADRADAR_ORDER_SIZES.slice()
    },

    getState() { return Object.assign({}, state); },
    modules: MODULES.slice()
  };

(function () {
  "use strict";

  if (window.__MIX_RUNTIME__?.version) return;

  const CARD_SELECTOR = ".trade-card";
  const SCHEDULER_TICK_MS = 100;
  const CATEGORY_INTERVALS = { fast: 350, normal: 600, slow: 1500 };

  const tasks = new Map();
  const variantGroups = Object.create(null);
  let schedulerTimer = 0;
  let schedulerRunning = false;

  function now() { return Date.now(); }

  function normalizeCategory(cat) {
    return Object.prototype.hasOwnProperty.call(CATEGORY_INTERVALS, cat) ? cat : "normal";
  }

  function normalizeInterval(options) {
    if (Number.isFinite(options?.interval) && options.interval > 0) return Math.max(50, Math.floor(options.interval));
    return CATEGORY_INTERVALS[normalizeCategory(options?.category)];
  }

  function runTask(task, ts) {
    try { task.run({ now: ts, runtime: api, task }); }
    catch (err) { console.warn("MIX runtime task failed:", task.id, err); }
  }

  function runDueTasks() {
    if (schedulerRunning) return;
    schedulerRunning = true;
    const ts = now();
    try {
      tasks.forEach(task => {
        if (task.paused || ts - task.lastRun < task.interval) return;
        task.lastRun = ts;
        runTask(task, ts);
      });
    } finally {
      schedulerRunning = false;
      if (!tasks.size && schedulerTimer) { clearInterval(schedulerTimer); schedulerTimer = 0; }
    }
  }

  function ensureScheduler() {
    if (!schedulerTimer) schedulerTimer = setInterval(runDueTasks, SCHEDULER_TICK_MS);
  }

  const scheduler = {
    register(id, options) {
      if (!id || !options || typeof options.run !== "function") return () => {};
      const task = {
        id,
        category: normalizeCategory(options.category),
        interval: normalizeInterval(options),
        run: options.run,
        lastRun: now(),
        paused: false
      };
      tasks.set(id, task);
      ensureScheduler();
      if (options.immediate) runTask(task, task.lastRun);
      return () => scheduler.unregister(id);
    },
    unregister(id) {
      tasks.delete(id);
      if (!tasks.size && schedulerTimer) { clearInterval(schedulerTimer); schedulerTimer = 0; }
    },
    runNow(id) {
      const task = tasks.get(id);
      if (task) { task.lastRun = now(); runTask(task, task.lastRun); }
    },
    pause(id)  { const t = tasks.get(id); if (t) t.paused = true; },
    resume(id) { const t = tasks.get(id); if (t) { t.paused = false; t.lastRun = now(); } },
    list() { return Array.from(tasks.values()).map(t => ({ id: t.id, category: t.category, interval: t.interval, paused: !!t.paused })); }
  };

    selector: CARD_SELECTOR,
    all(root = document) {
      const base = root?.querySelectorAll ? root : document;
      return Array.from(base.querySelectorAll(CARD_SELECTOR));

    forEach(fn, root = document) { if (typeof fn === "function") cards.all(root).forEach(fn); },
    isCard(node) { return node instanceof HTMLElement && node.matches(CARD_SELECTOR); },

      if (pending) return;

        run: () => { scheduler.unregister(FLUSH_TASK_ID); flush(); }
      });
    }

    function flush() {
      pending = false;
      const cardList = queuedCards.size
        ? Array.from(queuedCards).filter(c => document.body?.contains(c))
        : [];
      const mutationList = queuedMutations.length ? queuedMutations.splice(0) : [];
      queuedCards.clear();

      subscribers.forEach((handlers, id) => {
        try {
          if (cardList.length && typeof handlers.onCards === "function")
            handlers.onCards(cardList, { runtime: api, mutations: mutationList });
          if (typeof handlers.onFlush === "function")
            handlers.onFlush({ runtime: api, cards: cardList, mutations: mutationList });
        } catch (err) { console.warn("MIX card refresh subscriber failed:", id, err); }
      });
    }

    function handleMutations(mutations) {
      for (const mutation of mutations) {
        let relevant = queueFromNode(mutation.target) || isGlobalCardMetaNode(mutation.target);
        mutation.addedNodes?.forEach(node => {
          relevant = queueFromNode(node) || isGlobalCardMetaNode(node) || relevant;
        });
        if (relevant) queuedMutations.push(mutation);
      }
      if (queuedCards.size || queuedMutations.length) scheduleFlush();
    }

    function start() {
      if (observer) return;
      const target = document.body || document.documentElement;
      if (!target) return;
      observer = new MutationObserver(handleMutations);
      observer.observe(target, { childList: true, characterData: true, subtree: true });
    }

    function subscribe(id, handlers = {}) {
      if (!id) return () => {};
      subscribers.set(id, handlers);
      start();
      return () => subscribers.delete(id);

    normalizeExchange(raw) {

    },

    parseExchangePair(text) {
      if (!text) return null;
      const parts = String(text).replace(/[\d.,]+%/g, "-").split(/\s*[-–—]\s*/).map(p => p.trim()).filter(Boolean);
      if (parts.length < 2) return null;
      const first = helpers.normalizeExchange(parts[0]);
      const second = helpers.normalizeExchange(parts[1]);
      return first && second ? [first, second] : null;
    },

    parseExchangePairLoose(text, knownExchanges) {
      if (!text || !knownExchanges) return null;
      const names = Array.isArray(knownExchanges) ? knownExchanges : Object.keys(knownExchanges);
      if (!names.length) return null;
      const normalizedText = String(text).replace(/\d+(?:[.,]\d+)?\s*%/g, " ").replace(/[^A-Z0-9.\s]/gi, " ").toUpperCase().replace(/GATE\.IO/g, "GATEIO");
      const found = [];
      names.map(n => helpers.normalizeExchange(n)).filter(Boolean).sort((a, b) => b.length - a.length).forEach(name => {
        const escaped = name.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

        if (match) found.push({ name, index: match.index + match[1].length });
      });
      found.sort((a, b) => a.index - b.index);
      return found.length >= 2 ? [found[0].name, found[1].name] : null;
    },

    exchangePair(card, knownExchanges) {
      const text = helpers.arbitrageText(card);
      return helpers.parseExchangePair(text) || helpers.parseExchangePairLoose(text, knownExchanges);
    },

    coinBase(card) {
      let v = helpers.ticker(card).toUpperCase().replace(/[^A-Z0-9]/g, "");
      if (v.endsWith("USDT")) v = v.slice(0, -4);
      return v;
    },

    cardMeta(card, knownExchanges) {
      const coin = helpers.ticker(card);
      const arbitrageText = helpers.arbitrageText(card);
      return {
        coin,
        coinBase: helpers.coinBase(card),
        arbitrageText,
        exchanges: helpers.parseExchangePair(arbitrageText) || helpers.parseExchangePairLoose(arbitrageText, knownExchanges),
        side: helpers.side(card)
      };
    },

    findSaveButtonInGroup(group) {
      if (!group) return null;
      return Array.from(group.querySelectorAll?.("button") || []).find(btn => (btn.textContent || "").trim().toLowerCase() === "save") || null;
    },

    findTickControls(card) {

      if (movedOpen && movedClose) return { openInput: movedOpen, closeInput: movedClose, saveButtons: [movedOpenSave, movedCloseSave].filter(Boolean) };

      if (inputs.length < 2) return null;
      return { openInput: inputs[0], closeInput: inputs[1], saveButtons };
    },

    safeJsonParse(raw, fallback) {
      try { return raw ? JSON.parse(raw) : fallback; } catch (_) { return fallback; }
    }
  };

  const variants = {
    register(group, name, impl) {

  function pruneSamples(samples, now) {
    const maxMs = MAX_WINDOW_MIN * 60 * 1000 + 5000;
    while (samples.length && now - samples[0][0] > maxMs) samples.shift();
  }

  function hydrateState(key, now) {
    if (data.has(key)) return data.get(key);
    const s = persisted[key];
    if (s) {
      const d = { samples: Array.isArray(s.samples) ? s.samples : [], green: s.green, pendingAt: s.pendingAt || 0, firstSeenAt: s.firstSeenAt || now, g: s.g || 0 };
      pruneSamples(d.samples, now); data.set(key, d); return d;
    }
    const d = { samples: [], green: undefined, pendingAt: 0, firstSeenAt: now, g: 0 };
    data.set(key, d); return d;
  }

  function computeWindow(samples, now, windowMin) {
    const from = now - windowMin * 60 * 1000;
    let oMax = -Infinity, cMin = Infinity;
    for (let i = samples.length - 1; i >= 0; i--) {
      const s = samples[i]; if (s[0] < from) break;
      if (s[1] > oMax) oMax = s[1]; if (s[2] < cMin) cMin = s[2];
    }
    return (Number.isFinite(oMax) && Number.isFinite(cMin)) ? { oMax, cMin, R: oMax - cMin } : null;
  }

  function getLevelByRatio(R, fee) {
    if (!Number.isFinite(R) || !Number.isFinite(fee) || fee <= 0) return { level: "red" };
    const ratio = R / fee;
    return { level: ratio >= 3 ? "green" : ratio >= 2 ? "yellow" : "red", ratio };
  }

  function updateGreenState(d, isGreen, now) {
    if (d.green === undefined) { d.green = isGreen; d.pendingAt = 0; return false; }
    if (isGreen === d.green) { d.pendingAt = 0; return false; }
    if (!d.pendingAt) d.pendingAt = now;
    if (now - d.pendingAt < (isGreen ? ENTER_MS : EXIT_MS)) return false;
    d.green = isGreen; d.pendingAt = 0; return isGreen;
  }

  function updateCard(card) {
    if (!started || !isEnabled()) return;
    const key = getCardKey(card); if (!key) return;
    const oc  = getOpenClose(card); if (!oc) return;
    const now = Date.now();
    const d   = hydrateState(key, now);
    const last = d.samples[d.samples.length - 1];
    if (!last || now - last[0] >= SAMPLE_MS - 50) {
      d.samples.push([now, Number(oc.openRaw.toFixed(4)), Number(oc.closeRaw.toFixed(4))]);
      pruneSamples(d.samples, now); scheduleFlush();
    }
    const windowMin = windowPref[key] || 5;
    const res = computeWindow(d.samples, now, windowMin); if (!res) return;
    const fee = getFeeRound(card);
    const wrap = ensureUI(card);
    wrap.querySelectorAll(".mix-btn").forEach(b => b.classList.toggle("active", b.textContent === `${windowMin}m`));
    const summary = wrap.querySelector(".mix-summary");
    if (fee === null) {
      summary.textContent = `R:${fmt2(res.R)} | Open:${fmt2(res.oMax)} | Close:${fmt2(res.cMin)} | G:${d.g || 0} (${fmtAge(now - d.firstSeenAt)})`;
      setLevel(summary, "red"); return;
    }
    const lvl = getLevelByRatio(res.R, fee);
    setLevel(summary, lvl.level);
    const enteredGreen = updateGreenState(d, lvl.level === "green", now);
    if (enteredGreen) { d.g = (d.g || 0) + 1; scheduleFlush(); }
    summary.innerHTML = `R:<b>${fmt2(res.R)}</b> | Open:${fmt2(res.oMax)} | Close:${fmt2(res.cMin)} | G:<b>${d.g || 0}</b> (${fmtAge(now - d.firstSeenAt)})`;
  }

  function handleStopClick(e) {
    const btn = e.target.closest("button");
    if (!btn || (btn.textContent || "").trim().toLowerCase() !== "stop") return;
    const card = btn.closest(".trade-card"); if (!card) return;
    const key = getCardKey(card); if (!key) return;
    setRangeTimeout(() => {
      data.delete(key); delete persisted[key]; delete windowPref[key]; saveWindows();
      try { localStorage.setItem(LS_STATE, JSON.stringify(persisted)); } catch (_) {}
      const summary = card.querySelector(".mix-range-wrap .mix-summary");
      if (summary) { summary.textContent = "R:- | Open:- | Close:- | G:-"; summary.classList.remove("mix-level-red", "mix-level-yellow", "mix-level-green"); }
    }, 50);
  }

  function cleanupUi() { document.querySelectorAll(".mix-range-wrap").forEach(el => el.remove()); }

  function tick() {
    if (!started || !isEnabled()) { cleanupUi(); return; }
    injectStyle();
    if (!stopBound) { stopBound = true; document.addEventListener("click", handleStopClick, true); }
    (window.__MIX_RUNTIME__?.cards?.all() || Array.from(document.querySelectorAll(".trade-card"))).forEach(updateCard);
  }

  function start() {
    if (started) return;
    started = true; injectStyle();
    if (!stopBound) { stopBound = true; document.addEventListener("click", handleStopClick, true); }
    tick();
    if (scheduler) { scheduler.unregister?.(TASK_ID); scheduler.register(TASK_ID, { category: "normal", interval: SAMPLE_MS, run: tick }); }
    else if (!fallbackTimer) fallbackTimer = setInterval(tick, SAMPLE_MS);
  }

  function stop() {
    started = false;
    scheduler?.unregister?.(TASK_ID);
    if (fallbackTimer) { clearInterval(fallbackTimer); fallbackTimer = 0; }
    if (stopBound) { stopBound = false; document.removeEventListener("click", handleStopClick, true); }
    clearRangeTimeouts(); flushPendingState(); cleanupUi();
  }

  window.__MIX_MODULES__.onToggle(MODULE_ID, enabled => { if (enabled) start(); else stop(); });
  if (isEnabled()) start(); else stop();
})();

  function normalizeSymbol(raw, card) {
    const compact = String(raw || "").trim().replace(/\s+/g, "").replace(/[|:()[\]{}<>]/g, "");
    const direct = compact.toUpperCase();
    if (direct && /USDT$/.test(direct.replace(/_/g, ""))) return direct;
    const base = String(runtimeHelpers?.coinBase?.(card) || direct || "").trim().toUpperCase().replace(/[^A-Z0-9]/g, "");
    return base ? `${base}USDT` : "";
  }

  function normalizeExchange(raw) {
    const key = String(raw || "").replace(/[↑↓]/g, "").replace(/\s+/g, "").trim().toUpperCase().replace(/[^A-Z.]/g, "");
    return EXCHANGE_MAP[key] || "";
  }

  function cleanSignalTickerText(host) {
    if (!host) return "";
    const clone = host.cloneNode(true);
    clone.querySelectorAll?.(".mix-signal-live-inline").forEach(node => node.remove());
    return String(clone.textContent || "").trim();
  }

  function getRoute(card) {
    const ticker = normalizeSymbol(runtimeHelpers?.ticker?.(card) || card.querySelector(".trade-details strong")?.textContent || "", card);
    const longNode = card.querySelector(".arbitrage-info .long");
    const shortNode = card.querySelector(".arbitrage-info .short");
    let long = normalizeExchange(longNode?.textContent || "");
    let short = normalizeExchange(shortNode?.textContent || "");

    if (!long || !short) {
      const pair = runtimeHelpers?.exchangePair?.(card, EXCHANGE_MAP)
        || runtimeHelpers?.parseExchangePair?.(card.querySelector(".arbitrage-info")?.textContent || "")
        || runtimeHelpers?.parseExchangePairLoose?.(card.querySelector(".arbitrage-info")?.textContent || "", EXCHANGE_MAP);
      if (Array.isArray(pair) && pair.length >= 2) {
        long ||= normalizeExchange(pair[0]);
        short ||= normalizeExchange(pair[1]);
      }
    }

    return { ticker, long, short };
  }

  function getSignalRoute(card) {
    const text = String(card?.innerText || card?.textContent || "").replace(/\s+/g, " ").trim();
    const symbolText = cleanSignalTickerText(card?.querySelector?.('[data-help="signal.symbol"], td.symbol, .header .symbol, .symbol'));
    const ticker = normalizeSymbol(symbolText || (text.match(/([^\s|:()[\]{}<>]+_?USDT)/i) || [])[1] || "", null);
    const matches = Array.from(text.matchAll(/\b(BINANCE|MEXC|GATEIO|GATE\.IO|GATE|BINGX|BING|BYBITFUTURES|BYBITPERP|BYBIT\.USDT|BYBIT\.FUTURES|BYBIT|BITGET|ASTERDEX|ASTER\.DEX|ASTER)\b\s*([↑↓])/gi));
    let long = "";
    let short = "";

    for (const match of matches) {
      const exchange = normalizeExchange(match[1]);
      const arrow = match[2];
      if (!exchange) continue;
      if (arrow === "↑" && !long) long = exchange;
      if (arrow === "↓" && !short) short = exchange;
    }

    if ((!long || !short) && matches.length >= 2) {
      long ||= normalizeExchange(matches[0][1]);
      short ||= normalizeExchange(matches[1][1]);
    }

    return { ticker, long, short };
  }

  function buildScreenerUrl(route) {
    const params = new URLSearchParams();
    params.set("ticker", route.ticker);
    params.set("long", route.long);
    params.set("short", route.short);
    params.set("view", DEFAULT_VIEW);
    params.set("hours", DEFAULT_HOURS);
    params.set("type", DEFAULT_TYPE);
    return `${BASE_URL}?${params.toString()}`;
  }

  function formatRouteSymbol(ticker) {
    const symbol = normalizeSymbol(ticker, null);
    if (!symbol) return "";
    if (/_USDT$/i.test(symbol)) return symbol;
    if (/USDT$/i.test(symbol)) return `${symbol.slice(0, -4)}_USDT`;
    return `${symbol}_USDT`;
  }

  }

  }

  function getSpreadTone(value) {
    const spread = Number(value);
    if (!Number.isFinite(spread)) {
      return { className: "is-muted", color: "#94a3b8", fill: "rgba(148,163,184,.1)" };
    }

    if (spread >= 0.2) {
      const index = Math.min(Math.floor((spread - 0.2) / 0.1), SPREAD_TONE_PALETTE.posGreen.length - 1);
      return SPREAD_TONE_PALETTE.posGreen[Math.max(0, index)];
    }

    if (spread >= 0) {
      const index = Math.min(Math.floor(spread / 0.04), SPREAD_TONE_PALETTE.posWarm.length - 1);
      return SPREAD_TONE_PALETTE.posWarm[Math.max(0, index)];
    }

    const absSpread = Math.abs(spread);
    if (absSpread >= 0.2) {
      const index = Math.min(Math.floor((absSpread - 0.2) / 0.1), SPREAD_TONE_PALETTE.negDeep.length - 1);
      return SPREAD_TONE_PALETTE.negDeep[Math.max(0, index)];
    }

    const index = Math.min(Math.floor(absSpread / 0.04), SPREAD_TONE_PALETTE.negSoft.length - 1);
    return SPREAD_TONE_PALETTE.negSoft[Math.max(0, index)];
  }

  function renderSpreadChart(points) {
    const values = (Array.isArray(points) ? points : [])
      .filter((point) => point?.entry_order_depth_ok !== false)
      .slice(-10)
      .map((point) => Number(point?.entry_spread_pct))
      .filter((value) => Number.isFinite(value));
    if (values.length < 2) return "";

    const width = 220;
    const height = 34;
    const leftPad = 5;
    const rightPad = 5;
    const topPad = 7;
    const bottomPad = 7;
    const rawMin = Math.min(...values);
    const rawMax = Math.max(...values);
    let min = Math.min(0, rawMin);
    let max = Math.max(0, rawMax);
    let range = max - min;
    if (range < 0.001) {
      min -= 0.001;
      max += 0.001;
      range = max - min;
    }
    const padding = Math.max(range * 0.12, 0.01);
    min -= padding;
    max += padding;
    range = max - min;
    const xStep = values.length > 1 ? (width - leftPad - rightPad) / (values.length - 1) : 0;
    const yFor = (value) => {
      const ratio = (value - min) / range;
      return height - bottomPad - ratio * (height - topPad - bottomPad);
    };
    const linePoints = values
      .map((value, index) => `${(leftPad + index * xStep).toFixed(1)},${yFor(value).toFixed(1)}`)
      .join(" ");
    const last = values[values.length - 1];
    const tone = getSpreadTone(last);
    const lineColor = tone.color;
    const fillColor = tone.fill;
    const baselineY = yFor(0).toFixed(1);
    const areaPoints = `${leftPad},${baselineY} ${linePoints} ${width - rightPad},${baselineY}`;

    return `
      <div class="sr-live-chart" aria-hidden="true">
        <svg viewBox="0 0 ${width} ${height}" preserveAspectRatio="none">
          <line class="sr-chart-zero" x1="${leftPad}" y1="${baselineY}" x2="${width - rightPad}" y2="${baselineY}"></line>
          <polygon class="sr-chart-fill" points="${areaPoints}" style="fill:${fillColor}"></polygon>
          <polyline class="sr-chart-line" points="${linePoints}" style="stroke:${lineColor}"></polyline>
        </svg>
      </div>
    `;
  }

  async function pollLiveSpread() {
    if (!isEnabled() || !state.live.routeKey || state.live.inFlight) return;
    state.live.inFlight = true;
    state.live.controller?.abort?.();
    state.live.controller = new AbortController();

    try {
      const orderUsdt = getLiveOrderUsdt();
      const params = new URLSearchParams({
        route_key: state.live.routeKey,
        limit: "10",
        use_raw_depth: "1"
      });
      if (state.live.orderQtyBase) params.set("order_qty_base", String(state.live.orderQtyBase));
      const response = await fetch(`${LIVE_SPREAD_URL}?${params}`, {
        credentials: "include",
        signal: state.live.controller.signal
      });
      const data = await response.json();
      if (!response.ok) {
        if (response.status === 402 || data?.error === "subscription_required") {
          renderLiveTip('<div class="sr-live-muted">SpreadRadar login/subscription required</div>');
          return;
        }
        throw new Error(`HTTP ${response.status}`);
      }
      const points = Array.isArray(data?.points) ? data.points : [];
      const last = points[points.length - 1];
      if (!last) {
        renderLiveTip('<div class="sr-live-muted">No live data</div>');
        return;
      }
      if (!state.live.orderQtyBase) {
        const longAsk = Number(last.long_ask);
        if (Number.isFinite(longAsk) && longAsk > 0) {
          state.live.orderQtyBase = orderUsdt / longAsk;
          state.live.inFlight = false;
          return await pollLiveSpread();
        }
      }

      const orderLabel = formatUsdt(last.order_notional_usdt || orderUsdt);
      const isDepthNo = last.entry_order_depth_ok === false;
      const depthText = isDepthNo
        ? '<span class="sr-live-title">DEPTH:</span> <span class="sr-live-depth-no">NO</span>'
        : '<span class="sr-live-title">DEPTH:</span> <span class="sr-live-depth-ok">OK</span>';
      const openTone = getSpreadTone(last.entry_spread_pct);
      const openValueHtml = isDepthNo
        ? ""
        : `<span class="sr-live-spread sr-spread-tone ${openTone.className}">${formatPct(last.entry_spread_pct)}</span>`;
      const openRowHtml = `<div class="sr-live-row"><span><span class="sr-live-title">OPEN:</span> <span class="sr-live-money">${orderLabel}</span></span>${openValueHtml}</div>`;
      const chartHtml = renderSpreadChart(points);
      renderLiveTip(`
        ${openRowHtml}
        <div class="sr-live-row"><span class="sr-live-label">${depthText}</span><span class="sr-live-muted">${formatAge(last.ts_ms)}</span></div>
        ${chartHtml}
      `);
    } catch (err) {
      if (err?.name !== "AbortError") renderLiveTip('<div class="sr-live-muted">Live unavailable</div>');
    } finally {
      state.live.inFlight = false;
    }
  }

  function isSignalPanelActiveForInlineLive() {
    return document.body?.classList?.contains("mix-at-signal-controls-on")
      && !document.body?.classList?.contains("signals-hidden")
      && !document.body?.classList?.contains("mix-at-hide-main-ui")
      && window.innerWidth > 1200;
  }

  function isVisibleSignalCardInPanel(card, panel) {
    if (!(card instanceof HTMLElement) || !card.isConnected) return false;
    if (card.dataset.mixAtHidden === "true" || card.dataset.spreadradarHidden === "true") return false;

    const style = getComputedStyle(card);
    if (style.display === "none" || style.visibility === "hidden" || Number(style.opacity) === 0) return false;

    const rect = card.getBoundingClientRect();
    if (rect.width <= 0 || rect.height <= 0) return false;

    const panelRect = panel?.getBoundingClientRect?.();
    if (panelRect && panelRect.width > 0 && panelRect.height > 0) {
      return rect.bottom > panelRect.top && rect.top < panelRect.bottom && rect.right > panelRect.left && rect.left < panelRect.right;
    }

    return rect.bottom > 0 && rect.top < window.innerHeight && rect.right > 0 && rect.left < window.innerWidth;
  }

  function getInlineSignalCards() {
    if (!isSignalPanelActiveForInlineLive()) return [];
    const panel = document.querySelector(".signals-list");

    return Array.from(panel.querySelectorAll(".signal-card"))
      .filter(card => isVisibleSignalCardInPanel(card, panel))

  function getSignalTickerHost(card) {

  function getSignalInlineLiveHost(card) {
    if (!card) return null;

    const existingHost = card.querySelector?.("table.exchange-info td.mix-signal-live-host");
    if (existingHost) return existingHost;

    const row = card.querySelector?.("table.exchange-info tbody tr, table.exchange-info tr");

    const cleanText = (el) => {
      const clone = el.cloneNode(true);
      clone.querySelectorAll?.(".mix-signal-live-inline").forEach(node => node.remove());

    const emptyCell = cells.find(el => !el.classList.contains("short") && !el.classList.contains("long") && cleanText(el) === "");
    return emptyCell || getSignalTickerHost(card);

  function ensureInlineLiveNode(card) {
    const host = getSignalInlineLiveHost(card);
    if (!host) return null;
    host.classList.add("mix-signal-live-host");

    const node = ensureInlineLiveNode(card);
    if (!node) return;
    node.className = `mix-signal-live-inline ${className || "is-muted"}`;
    node.textContent = text;
    node.dataset.mixInlineLiveUpdatedAt = String(Date.now());
    if (routeKey) node.dataset.mixInlineLiveRouteKey = routeKey;

  function removeInlineLiveNode(card) {
    const host = getSignalInlineLiveHost(card);
    if (!host) return;
    host.querySelectorAll?.(".mix-signal-live-inline").forEach(node => node.remove());
    host.classList.remove("mix-signal-live-host");

  function clearInlineLiveNodes(exceptCards = new Set(), force = false) {
    document.querySelectorAll(".mix-signal-live-inline").forEach(node => {
      const card = node.closest?.(".signal-card");
      if (card && exceptCards.has(card)) return;
      const host = node.parentElement;

      host?.classList?.remove("mix-signal-live-host");
    });

  function getCachedInlineLive(routeKey) {
    const item = state.inlineLive.lastByRoute?.[routeKey];
    if (!item || !Number.isFinite(Number(item.expiresAt)) || Date.now() > Number(item.expiresAt)) return null;
    return item;
  }

  function cacheInlineLive(routeKey, text, className) {
    if (!routeKey) return;
    state.inlineLive.lastByRoute[routeKey] = {
      text,
      className,
      expiresAt: Date.now() + INLINE_LIVE_TTL_MS
    };
  }

  async function fetchInlineLiveSpread(routeKey) {
    const orderUsdt = getLiveOrderUsdt();
    const qtyKey = `${routeKey}|${orderUsdt}`;
    let orderQtyBase = Number(state.inlineLive.orderQtyByRoute?.[qtyKey]);
    if (!Number.isFinite(orderQtyBase) || orderQtyBase <= 0) orderQtyBase = 0;

    for (let attempt = 0; attempt < 2; attempt += 1) {
      const params = new URLSearchParams({
        route_key: routeKey,
        limit: "1",
        use_raw_depth: "1"
      });
      if (orderQtyBase > 0) params.set("order_qty_base", String(orderQtyBase));

      const response = await fetch(`${LIVE_SPREAD_URL}?${params}`, { credentials: "include" });
      const data = await response.json().catch(() => null);
      if (!response.ok) throw new Error(data?.error || `HTTP ${response.status}`);

      const points = Array.isArray(data?.points) ? data.points : [];
      const last = points[points.length - 1];
      if (!last) return { status: "empty" };

      if (orderQtyBase <= 0) {
        const longAsk = Number(last.long_ask);
        if (Number.isFinite(longAsk) && longAsk > 0) {
          orderQtyBase = orderUsdt / longAsk;
          state.inlineLive.orderQtyByRoute[qtyKey] = orderQtyBase;
          continue;
        }
      }

      if (last.entry_order_depth_ok === false) return { status: "depth_no" };
      const spread = Number(last.entry_spread_pct);
      if (!Number.isFinite(spread)) return { status: "empty" };
      return { status: "ok", spread };
    }

    return { status: "empty" };
  }

  async function updateInlineLiveSpreads() {
    if (!isEnabled() || state.inlineLive.inFlight) return;
    const cards = getInlineSignalCards();
    const cardSet = new Set(cards);
    clearInlineLiveNodes(cardSet);
    if (!cards.length) return;

    state.inlineLive.inFlight = true;
    try {
      for (const card of cards) {
        const routeKey = getSignalRouteKey(card);
        if (!routeKey) {
          setInlineLiveNode(card, "LIVE —", "is-muted");
          continue;
        }

        const cached = getCachedInlineLive(routeKey);
        if (cached) setInlineLiveNode(card, cached.text, cached.className, routeKey);
        else {
          const node = ensureInlineLiveNode(card);
          if (node && (!node.textContent || node.textContent === "LIVE —")) {
            node.textContent = "LIVE …";
            node.className = "mix-signal-live-inline is-muted";
          }
        }

        try {
          const result = await fetchInlineLiveSpread(routeKey);
          if (result.status === "depth_no") {
            removeInlineLiveNode(card);
            continue;
          }
          if (result.status !== "ok") {
            const latestCached = getCachedInlineLive(routeKey);
            if (latestCached) setInlineLiveNode(card, latestCached.text, latestCached.className, routeKey);
            else setInlineLiveNode(card, "LIVE —", "is-muted", routeKey);
            continue;
          }
          const cls = `sr-spread-tone ${getSpreadTone(result.spread).className}`;
          const text = formatPct(result.spread);
          cacheInlineLive(routeKey, text, cls);
          setInlineLiveNode(card, text, cls, routeKey);
        } catch (_) {
          const latestCached = getCachedInlineLive(routeKey);
          if (latestCached) setInlineLiveNode(card, latestCached.text, latestCached.className, routeKey);
          else setInlineLiveNode(card, "LIVE —", "is-muted", routeKey);
        }
      }
    } finally {
      state.inlineLive.inFlight = false;
    }
  }

  function startInlineLiveSpreads() {
    if (state.inlineLive.timer) return;
    updateInlineLiveSpreads();
    state.inlineLive.timer = setInterval(updateInlineLiveSpreads, INLINE_LIVE_POLL_MS);
  }

  function stopInlineLiveSpreads() {
    if (state.inlineLive.timer) clearInterval(state.inlineLive.timer);
    state.inlineLive.timer = 0;
    state.inlineLive.inFlight = false;
    clearInlineLiveNodes(new Set(), true);
  }

  function startLiveHover(tickerEl) {
    const card = tickerEl.closest(".signal-card");
    if (!card) return;
    const routeKey = getSignalRouteKey(card);
    if (!routeKey) return;
    if (state.live.tickerEl === tickerEl && state.live.routeKey === routeKey) return;

    stopLiveHover();
    state.live.tickerEl = tickerEl;
    state.live.card = card;
    state.live.routeKey = routeKey;
    renderLiveTip('<div class="sr-live-muted">Loading live spread...</div>');
    pollLiveSpread();
    state.live.timer = setInterval(pollLiveSpread, LIVE_POLL_MS);
  }

    if (!isEnabled()) return;
    const ticker = getSignalTickerFromEventTarget(event.target);
    if (ticker) startLiveHover(ticker);
  }

  function onSignalTickerHoverOut(event) {
    const ticker = getSignalTickerFromEventTarget(event.target);
    if (!ticker) return;
    const next = event.relatedTarget;
    if (next instanceof Node && ticker.contains(next)) return;
    stopLiveHover();
  }

  function onSignalTickerClick(event) {
    if (!isEnabled()) return;
    const ticker = getSignalTickerFromEventTarget(event.target);
    if (!ticker) return;
    const card = ticker.closest(".signal-card");
    if (!card) return;
    event.preventDefault();
    event.stopPropagation();
    openSignalScreener(card);
  }

    state.signalClickHandler = onSignalTickerClick;
    document.addEventListener("click", state.signalClickHandler, true);
  }

  function detachSignalDelegatedClick() {
    if (!state.signalClickHandler) return;
    document.removeEventListener("click", state.signalClickHandler, true);
    state.signalClickHandler = null;
  }

  function attachSignalHover() {
    if (!state.signalHoverInHandler) {
      state.signalHoverInHandler = onSignalTickerHoverIn;
      document.addEventListener("mouseover", state.signalHoverInHandler, true);
    }
    if (!state.signalHoverOutHandler) {
      state.signalHoverOutHandler = onSignalTickerHoverOut;
      document.addEventListener("mouseout", state.signalHoverOutHandler, true);
    }
  }

  function detachSignalHover() {
    if (state.signalHoverInHandler) {
      document.removeEventListener("mouseover", state.signalHoverInHandler, true);
      state.signalHoverInHandler = null;
    }
    if (state.signalHoverOutHandler) {
      document.removeEventListener("mouseout", state.signalHoverOutHandler, true);
      state.signalHoverOutHandler = null;
    }
    stopLiveHover();
  }

  function onSpreadRadarSettingsChanged() {
    state.live.orderQtyBase = null;
    state.inlineLive.orderQtyByRoute = Object.create(null);
    state.inlineLive.lastByRoute = Object.create(null);
    if (state.live.routeKey) pollLiveSpread();
    updateInlineLiveSpreads();
  }

  function stop() {
    state.observer?.disconnect?.(); state.observer = null;
    if (typeof state.unsubscribeRefresh === "function") { state.unsubscribeRefresh(); state.unsubscribeRefresh = null; }
    if (state.delayedTimer) { clearTimeout(state.delayedTimer); state.delayedTimer = 0; }
    detachSignalDelegatedClick();
    detachSignalHover();
    stopInlineLiveSpreads();
    document.removeEventListener("mix:spreadradar-settings-changed", onSpreadRadarSettingsChanged);
    document.body?.classList?.remove("mix-spread-screener-on");
    document.querySelectorAll(".mix-spread-screener-ticker").forEach(el => {
      el.classList.remove("mix-spread-screener-ticker");
      if (el.title === "Open spread history screener") el.removeAttribute("title");
    });
    document.querySelectorAll(".mix-spread-screener-signal-ticker").forEach(el => {
      el.classList.remove("mix-spread-screener-signal-ticker");
      if (el.title === "Open spread history screener") el.removeAttribute("title");
    });
  }

  function start() {
    if (!isEnabled()) { stop(); return; }
    document.body?.classList?.add("mix-spread-screener-on");
    document.removeEventListener("mix:spreadradar-settings-changed", onSpreadRadarSettingsChanged);
    document.addEventListener("mix:spreadradar-settings-changed", onSpreadRadarSettingsChanged);
    attachSignalDelegatedClick();
    attachSignalHover();
    bindAll();
    startInlineLiveSpreads();
    if (cardRefresh && !state.unsubscribeRefresh) {
      state.unsubscribeRefresh = cardRefresh.subscribe("spreadScreener", {
        onCards(cardList) { if (isEnabled()) cardList.forEach(bindCard); }
      });
    }
    if (!state.observer && document.body) {
      state.observer = new MutationObserver(mutations => {
        if (!isEnabled()) return;
        for (const mutation of mutations) {
          for (const node of mutation.addedNodes) {
            if (!(node instanceof HTMLElement)) continue;
            if (node.matches?.(".trade-card")) bindCard(node);
            node.querySelectorAll?.(".trade-card").forEach(bindCard);
            if (node.matches?.(".signal-card") || node.querySelector?.(".signal-card")) {
              updateInlineLiveSpreads();
            }
          }
        }
      });
      state.observer.observe(document.body, { childList: true, subtree: true });
    }
    if (state.delayedTimer) clearTimeout(state.delayedTimer);
    state.delayedTimer = setTimeout(() => { state.delayedTimer = 0; if (isEnabled()) { bindAll(); updateInlineLiveSpreads(); } }, 1500);
  }

  window.__MIX_MODULES__.onToggle(MODULE_ID, enabled => { if (enabled) start(); else stop(); });
  if (isEnabled()) start();
})();

/* ===== МОДУЛЬ: латиниця у полі Symbol і крапка у Spread ===== */

(function () {
  "use strict";

  const MODULE_ID = "tickerLatin";
  const STYLE_ID = "mix-ticker-latin-style-v1";
  const isEnabled = () => window.__MIX_MODULES__?.isEnabled?.(MODULE_ID) !== false;
  const updating = new WeakSet();
  const state = window.__MIX_TICKER_LATIN__ = window.__MIX_TICKER_LATIN__ || { started: false };

  const LAYOUT_MAP = Object.freeze({
    "й": "q", "ц": "w", "у": "e", "к": "r", "е": "t", "н": "y", "г": "u", "ш": "i", "щ": "o", "з": "p",
    "х": "[", "ї": "]", "ъ": "]",
    "ф": "a", "і": "s", "ы": "s", "в": "d", "а": "f", "п": "g", "р": "h", "о": "j", "л": "k", "д": "l",
    "ж": ";", "є": "'", "э": "'",
    "я": "z", "ч": "x", "с": "c", "м": "v", "и": "b", "т": "n", "ь": "m", "б": ",", "ю": ".", "ґ": "`", "ё": "`"
  });

  function injectStyle() {
    if (document.getElementById(STYLE_ID)) return;
    const style = document.createElement("style");
    style.id = STYLE_ID;
    style.textContent = `
      input.mix-ticker-latin-input{
        text-transform:uppercase!important;
        letter-spacing:.2px!important;
      }
    `;
    document.head.appendChild(style);
  }

  function isSymbolInput(target) {
    if (!(target instanceof HTMLInputElement)) return false;
    if (target.closest(".trade-card")) return false;
    const placeholder = String(target.getAttribute("placeholder") || "").trim().toLowerCase();
    return (
      placeholder === "symbol" &&
      (target.closest("form.new-trade-form") || target.closest('[data-help="createForm.symbol"]'))
    );
  }

  function isCreateFormSpreadInput(target) {
    if (!(target instanceof HTMLInputElement)) return false;
    if (target.closest(".trade-card")) return false;
    if (!target.closest("form.new-trade-form")) return false;
    const group = target.closest(".field-group");
    const labelText = String(group?.querySelector("label")?.textContent || "").replace(/\s+/g, " ").trim().toLowerCase();
    return labelText === "open spread:" || labelText === "close spread:";
  }

  function isTradeCardSpreadInput(target) {
    if (!(target instanceof HTMLInputElement)) return false;
    const card = target.closest(".trade-card");
    if (!card) return false;
    const group = target.closest(".field-group") || target.parentElement;
    const labelText = String(group?.querySelector("label,strong,b")?.textContent || "").replace(/\s+/g, " ").trim().toLowerCase();
    if (labelText === "open spread:" || labelText === "close spread:") return true;
    const groupText = String(group?.textContent || "").replace(/\s+/g, " ").trim().toLowerCase();
    return /^(open|close)\s+spread\s*:/.test(groupText);
  }

  function normalizeTickerValue(value) {
    return String(value || "")
      .split("")
      .map(ch => {
        const lower = ch.toLowerCase();
        return LAYOUT_MAP[lower] || ch;
      })
      .join("")
      .toUpperCase()
      .replace(/[^A-Z0-9]/g, "");
  }

  function normalizeSpreadValue(value) {
    return String(value || "").replace(/[бБюЮ/]/g, ".");
  }

  function setNativeValue(input, value) {
    const setter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value")?.set;
    if (setter) setter.call(input, value); else input.value = value;
  }

  function convertInput(input) {
    if (!isEnabled() || updating.has(input)) return;
    const isTicker = isSymbolInput(input);
    const isSpread = isCreateFormSpreadInput(input) || isTradeCardSpreadInput(input);
    if (!isTicker && !isSpread) return;
    const current = input.value;
    const next = isTicker ? normalizeTickerValue(current) : normalizeSpreadValue(current);
    if (isTicker) input.classList.add("mix-ticker-latin-input");
    if (next === current) return;

    updating.add(input);
    setNativeValue(input, next);
    input.dispatchEvent(new Event("input", { bubbles: true }));
    input.dispatchEvent(new Event("change", { bubbles: true }));
    try { input.setSelectionRange(next.length, next.length); } catch (_) {}
    requestAnimationFrame(() => updating.delete(input));
  }

  function onInput(event) {
    convertInput(event.target);
  }

  function onPaste(event) {
    if (!isSymbolInput(event.target) && !isCreateFormSpreadInput(event.target) && !isTradeCardSpreadInput(event.target)) return;
    requestAnimationFrame(() => convertInput(event.target));
  }

  function markExistingInputs() {
    document.querySelectorAll('form.new-trade-form input[placeholder="Symbol"], [data-help="createForm.symbol"] input[placeholder="Symbol"]').forEach(input => {
      if (input instanceof HTMLInputElement) {
        input.classList.add("mix-ticker-latin-input");
        convertInput(input);
      }
    });
  }

  function start() {
    if (state.started) return;
    state.started = true;
    injectStyle();
    document.addEventListener("input", onInput, true);
    document.addEventListener("paste", onPaste, true);
    markExistingInputs();
  }

  function stop() {
    if (!state.started) return;
    state.started = false;
    document.removeEventListener("input", onInput, true);
    document.removeEventListener("paste", onPaste, true);
    document.querySelectorAll(".mix-ticker-latin-input").forEach(input => input.classList.remove("mix-ticker-latin-input"));
  }

  window.__MIX_MODULES__.onToggle(MODULE_ID, enabled => { if (enabled) start(); else stop(); });
  if (isEnabled()) start();
})();

/* ===== МОДУЛЬ: множник Order Size (x2 / ÷2) ===== */

  function setReactValue(el, val) {

    if (setter) setter.call(el, val); else el.value = val;
    el.dispatchEvent(new Event("input",  { bubbles: true }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
  }

  function findOrderSizeInput(card) {
    for (const label of card.querySelectorAll("label")) {
      if (!/order size/i.test(label.textContent)) continue;
      const group = label.closest(".field-group") || label.parentElement || label.closest("div");
      const input = group?.querySelector?.('input[type="text"], input[type="number"], input');
      if (input) return input;
    }
    return null;
  }

  // Округлює "≈ X.XX $" у батьківській групі input
  // Шукає спочатку за відомим класом, потім по text-siblings
  function roundApproxInGroup(input) {
    const group = input.parentElement;
    if (!group) return;

    // Варіант 1: відомий клас .estimated-size-usdt
    const approxEl = group.querySelector(".estimated-size-usdt");
    if (approxEl) {
      const text = approxEl.textContent || "";
      const match = text.match(/≈\s*([\d.,]+)\s*\$/);
      if (match && (match[1].includes(".") || match[1].includes(","))) {
        const rounded = Math.round(parseFloat(match[1].replace(/,/g, "")));
        approxEl.textContent = text.replace(match[0], `≈ ${rounded} $`);
      }
      return;
    }

    // Варіант 2: текстові вузли та елементи після input до Save
    let curr = input.nextSibling;
    while (curr) {
      if (curr.nodeType === Node.ELEMENT_NODE &&
          curr.tagName === "BUTTON" &&
          /save/i.test(curr.textContent)) break;

      const text = curr.textContent || "";
      const match = text.match(/≈\s*([\d.,]+)\s*\$/);
      if (match && (match[1].includes(".") || match[1].includes(","))) {
        const rounded = Math.round(parseFloat(match[1].replace(/,/g, "")));
        curr.textContent = text.replace(match[0], `≈ ${rounded} $`);
      }
      curr = curr.nextSibling;
    }
  }

  // Окремий MutationObserver тільки на parentElement input.
  // Відключається перед своєю DOM-мутацією → не викликає себе рекурсивно.
  // processCard при mixOsmBound===1 робить лише return → цикл з cardRefresh розірвано.
  function bindApproxObserver(input) {
    const group = input.parentElement;
    if (!group || input._mixApproxObserver) return;

    let busy = false;

    const mo = new MutationObserver(() => {
      if (busy) return;
      busy = true;
      mo.disconnect();
      roundApproxInGroup(input);
      mo.observe(group, { subtree: true, characterData: true, childList: true });
      busy = false;
    });

    mo.observe(group, { subtree: true, characterData: true, childList: true });
    input._mixApproxObserver = mo;

    // Округлити одразу при підключенні
    roundApproxInGroup(input);
  }

  function processCard(card) {
    if (!isEnabled()) return;

    const input = findOrderSizeInput(card);
    if (!input) return;

    // Вже оброблено — нуль DOM-операцій, нуль ризику циклу
    // Округленням займається bindApproxObserver незалежно
    if (input.dataset.mixOsmBound === "1") return;

    input.dataset.mixOsmBound = "1";

    if (input.parentElement) input.parentElement.style.whiteSpace = "nowrap";
    input.style.width = "60px";
    input.style.textAlign = "center";

    const btnDiv2 = document.createElement("button");
    btnDiv2.type = "button";
    btnDiv2.className = "mix-osm-btn";
    btnDiv2.textContent = "÷2";
    btnDiv2.title = "Order Size ÷ 2";
    btnDiv2.addEventListener("click", e => {
      e.preventDefault();
      e.stopPropagation();
      const val = parseFloat(input.value) || 0;
      setReactValue(input, Number((val / 2).toFixed(6)));
    });

    const btnMul2 = document.createElement("button");
    btnMul2.type = "button";
    btnMul2.className = "mix-osm-btn";
    btnMul2.textContent = "x2";
    btnMul2.title = "Order Size × 2";
    btnMul2.addEventListener("click", e => {
      e.preventDefault();
      e.stopPropagation();
      const val = parseFloat(input.value) || 0;
      setReactValue(input, Number((val * 2).toFixed(6)));
    });

    input.after(btnDiv2);
    btnDiv2.after(btnMul2);

    // Підключаємо observer після вставки кнопок
    bindApproxObserver(input);
  }

  function cleanupCard(card) {
    card.querySelectorAll(".mix-osm-btn").forEach(btn => btn.remove());
    const input = findOrderSizeInput(card);
    if (!input) return;
    if (input._mixApproxObserver) {
      input._mixApproxObserver.disconnect();
      delete input._mixApproxObserver;
    }
    delete input.dataset.mixOsmBound;
    input.style.removeProperty("width");
    input.style.removeProperty("text-align");
    if (input.parentElement) input.parentElement.style.removeProperty("white-space");
  }

  function initAll() {
    injectStyle();
    const cardList = window.__MIX_RUNTIME__?.cards?.all()
      || Array.from(document.querySelectorAll(".trade-card"));
    cardList.forEach(processCard);
  }

  function start() {
    injectStyle();
    initAll();
    if (cardRefresh && !unsubscribeCardRefresh) {
      unsubscribeCardRefresh = cardRefresh.subscribe("orderSizeMultiplier", {
        onCards(cardList) {
          if (!isEnabled()) return;
          cardList.forEach(processCard);
        }
      });
    }
  }

  function stop() {
    if (typeof unsubscribeCardRefresh === "function") {
      unsubscribeCardRefresh();
      unsubscribeCardRefresh = null;
    }
    const cardList = window.__MIX_RUNTIME__?.cards?.all()
      || Array.from(document.querySelectorAll(".trade-card"));
    cardList.forEach(cleanupCard);
  }

  window.__MIX_MODULES__.onToggle(MODULE_ID, enabled => {
    if (enabled) start();
    else stop();
  });

  if (isEnabled()) start();
  else stop();
})();

  function normalizeCardIdentityText(value) {
    return String(value || "")
      .replace(/[\d]+[.,]?[\d]*\s*%/g, " ")
      .replace(/\b\d{1,2}:\d{2}(?::\d{2})?\b/g, " ")
      .replace(/\b(?:age|updated|orders|volume|funding|open|close|depth|yes|no|ok|last|price|size)\b/gi, " ")
      .replace(/[^A-Z0-9._-]+/gi, " ")
      .replace(/\s+/g, " ")
      .trim()
      .toUpperCase();
  }

  function normalizeCardExchange(value) {
    return String(value || "")
      .replace(/[↑↓]/g, "")
      .replace(/\s+/g, "")
      .trim()
      .toUpperCase()
      .replace(/[^A-Z0-9.]/g, "")
      .replace(/^GATE\.IO$/, "GATEIO");
  }

  function getSignalCardDomRouteId(card) {
    if (!card?.matches?.(".signal-card")) return "";

    const headerText = String(card.querySelector?.(".header")?.textContent || "");
    const longText = String(card.querySelector?.('[data-help="signal.exchanges"] td.long')?.textContent || "");
    const shortText = String(card.querySelector?.('[data-help="signal.exchanges"] td.short')?.textContent || "");
    const long = normalizeCardExchange(longText);
    const short = normalizeCardExchange(shortText);

    const headerSymbol = (headerText.match(/\b([A-Z0-9]{2,}[_\-/]?USDT)\b/i) || [])[1] || "";
    const symbol = headerSymbol
      ? headerSymbol.replace(/[\-/]/g, "_").toUpperCase()
      : normalizeCardIdentityText(headerText);

    if (symbol && long && short) return `signal:${symbol}:${long}:${short}`;
    if (long && short) return `signal-ex:${long}:${short}`;
    if (symbol) return `signal-header:${symbol}`;
    return "";
  }

  function getStableSignalCardId(card) {
    if (!card) return null;

    const direct =
      card.getAttribute?.("data-route-key") ||

      card.getAttribute?.("data-id") ||
      card.querySelector?.("input[type='hidden']")?.value ||
      "";

    if (String(direct).trim()) return `direct:${String(direct).trim()}`;

    const signalDomId = getSignalCardDomRouteId(card);
    if (signalDomId) return signalDomId;

    const meta = helpers?.cardMeta?.(card);
    const coin = String(meta?.coin || helpers?.ticker?.(card) || "").trim().toUpperCase();
    const pair = Array.isArray(meta?.exchanges) ? meta.exchanges.map(x => String(x || "").trim().toUpperCase()).filter(Boolean) : [];
    if (coin && pair.length >= 2) return `route:${coin}:${pair[0]}:${pair[1]}`;

    const normalizedRouteText = normalizeCardIdentityText(meta?.arbitrageText || helpers?.arbitrageText?.(card) || "");

    if (coin && normalizedRouteText) return `text:${coin}:${normalizedRouteText}`;

  function primeOrdersBaseline() {
    state.lastOrdersCount = {};
    state.knownCardIds = new Set();

    getAllSoundCards().forEach(card => {
      const snap = getOrderSnapshot(card);
      if (!snap?.cardId) return;
      state.lastOrdersCount[snap.cardId] = snap.orders;

    getAllSignalSoundCards().forEach(card => {
      const snap = getOrderSnapshot(card);
      if (!snap?.cardId) return;
      state.knownCardIds.add(snap.cardId);
    });
  }

  function maybeEmitCardsAppeared() {
    const now = Date.now();
    const currentIds = new Set();
    let newCards = 0;

    getAllSignalSoundCards().forEach(card => {
      const snap = getOrderSnapshot(card);
      if (!snap?.cardId) return;
      currentIds.add(snap.cardId);
      if (!state.knownCardIds.has(snap.cardId)) newCards += 1;
    });

    state.knownCardIds = currentIds;
    if (newCards <= 0) return;

    if (!state.lastCardSoundAt || now - state.lastCardSoundAt >= CARD_SOUND_BUFFER_MS) {
      state.lastCardSoundAt = now;
      emitSound("card");
    }
  }

  function getFreshCards(cardList) {
    if (Array.isArray(cardList) && cardList.length) {
      const filtered = cardList.filter(card => card instanceof HTMLElement && card.matches?.(ORDER_SOUND_CARD_SELECTOR));
      if (filtered.length) return filtered;

    return getAllSoundCards();
  }

  function tickOrders(cardList) {
    if (!isEnabled() || state.recoveryScheduled) return;
    maybeEmitCardsAppeared(cardList);
    getFreshCards(cardList).forEach(card => {
      const snap = getOrderSnapshot(card);
      const { cardId, orders: curOrders } = snap;
      if (!cardId) return;
      const prevOrders = state.lastOrdersCount[cardId];
      if (prevOrders !== undefined && curOrders !== prevOrders) {
        const ignore = getActiveIgnore(cardId);
        if (ignore) { state.ignoreNextOrderDeltaByCardId.delete(cardId); state.lastOrdersCount[cardId] = curOrders; return; }
        if (curOrders > prevOrders) { state.lastOpenAt = Date.now(); emitSound("open"); }
        else emitSound("close");
      }
      state.lastOrdersCount[cardId] = curOrders;
    });
  }

  function reconcileAfterVisibilityReturn(cardList) {
    if (!state.running || !isEnabled()) return;
    let openEmitted = false, closeEmitted = false;
    getFreshCards(cardList).forEach(card => {
      const snap = getOrderSnapshot(card);
      const { cardId, orders: curOrders } = snap;
      if (!cardId) return;
      const prevOrders = state.lastOrdersCount[cardId];
      if (prevOrders !== undefined && curOrders !== prevOrders) {
        const ignore = getActiveIgnore(cardId);
        if (ignore) { state.ignoreNextOrderDeltaByCardId.delete(cardId); state.lastOrdersCount[cardId] = curOrders; return; }
        if (curOrders > prevOrders && !openEmitted) { state.lastOpenAt = Date.now(); emitSound("open"); openEmitted = true; }
        else if (curOrders < prevOrders && !closeEmitted) { emitSound("close"); closeEmitted = true; }
      }
      state.lastOrdersCount[cardId] = curOrders;
    });
  }

  function stop() {

    scheduler?.unregister?.("sounds.orders");
    scheduler?.unregister?.("sounds.notifyFinder");
    unsubscribeCardRefresh();
    detachVisibilityListener(); detachPageShowListener();
    if (state.recoveryTimer) { clearTimeout(state.recoveryTimer); state.recoveryTimer = 0; }
    state.recoveryScheduled = false; state.lastVisibilityReturnAt = 0; state.burstSuppressUntil = 0;
    if (state.notifyObserver) { state.notifyObserver.disconnect(); state.notifyObserver = null; }

    if (state.recoveryTimer) { clearTimeout(state.recoveryTimer); state.recoveryTimer = 0; }
    scheduler.register("sounds.orders",      { category:"fast", interval:ORDER_SCAN_MS,  run:tickOrders });
    scheduler.register("sounds.notifyFinder", { category:"fast", interval:NOTIFY_FIND_MS, run:initNotifyByCount });
  }

  state.stop = stop; state.start = start;

  }

    const percent = Math.min(100, Math.max(0, Number(settings.feeCashbackPercent) || 0));
    return fee * percent / 100;
  }

  function setReactValue(el, val) {
    if (!el) return;
    const inputSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value")?.set;
    const selectSetter = Object.getOwnPropertyDescriptor(window.HTMLSelectElement.prototype, "value")?.set;
    if (el.tagName === "SELECT" && selectSetter) selectSetter.call(el, val);
    else if (inputSetter) inputSetter.call(el, val);
    else el.value = val;
    el.dispatchEvent(new Event("input", { bubbles: true }));
    el.dispatchEvent(new Event("change", { bubbles: true }));
  }

  function normalizeExchangeLabel(raw) {
    const text = String(raw || "").toUpperCase();
    const compact = text.replace(/[^A-Z0-9]/g, "");
    if (compact.includes("MEXC")) return "MEXC";
    if (compact.includes("BINGX") || compact.includes("BING")) return "BINGX";
    if (compact.includes("GATE")) return "GATE";
    if (compact.includes("BYBIT")) return "BYBIT";
    if (compact.includes("BITGET")) return "BITGET";
    if (compact.includes("BINANCE")) return "BINANCE";
    if (compact.includes("ASTERDEX") || compact.includes("ASTER")) return "ASTERDEX";
    if (compact.includes("KUCOIN")) return "KUCOIN";
    const match = text.match(/[A-Z0-9]+/);
    return match ? match[0] : text;
  }

  function parseNumberFromText(str) {
    if (!str) return NaN;
    return parseFloat(String(str).replace(/[^\d\-.,]/g, "").replace(/,/g, "."));
  }

  function collectTradeCardLimitsByExchange() {
    const limits = {};
    const limitHosts = $$('[data-help="cardLimits"], h4[data-help="cardLimits"]');
    const limitTextItems = [];

    limitHosts.forEach(host => {
      $$(".bots-limit-info", host).forEach(el => limitTextItems.push(el.textContent || ""));
      if (!host.classList?.contains("bots-limit-info")) limitTextItems.push(host.textContent || "");
    });

    limitTextItems.forEach(text => {
      const normalized = String(text || "").replace(/\s+/g, " ").trim();
      if (!normalized) return;
      const re = /([A-Z][A-Z0-9. _-]{1,18})\s*:\s*(\d+)\s*\/\s*(\d+)/gi;
      let match;
      while ((match = re.exec(normalized))) {
        const ex = normalizeExchangeLabel(match[1]);
        const used = Number(match[2]);
        const total = Number(match[3]);
        if (!ORDER.includes(ex) || !Number.isFinite(used) || !Number.isFinite(total) || total <= 0) continue;
        limits[ex] = { used, total };
      }
    });
    return limits;
  }

  function sumHistoryPnlAndFee(historyData) {
    const result = { pnl: 0, fee: 0, hasData: false };
    if (!historyData || typeof historyData !== "object") return result;

    const addPosition = position => {
      if (!position || typeof position !== "object") return;
      const pnl = Number(position.realisedPNL ?? position.realizedPNL ?? position.realisedPnl ?? position.realizedPnl ?? 0);
      const fee = Number(position.tradingFee ?? position.tradingfee ?? position.trading_fee ?? 0);
      if (Number.isFinite(pnl) || Number.isFinite(fee)) {
        result.hasData = true;
        result.pnl += Number.isFinite(pnl) ? pnl : 0;
        result.fee += Number.isFinite(fee) ? fee : 0;
      }
    };

    Object.entries(historyData).forEach(([exchange, item]) => {
      if (exchange === "success" || !Array.isArray(item) || item[0] !== true) return;
      (Array.isArray(item[1]) ? item[1] : []).forEach(addPosition);
      (Array.isArray(item[2]) ? item[2] : []).forEach(group => {
        if (!group || group.key === "-" || group.key === "undefined" || !Array.isArray(group.positions)) return;
        group.positions.forEach(addPosition);
      });
    });
    if (result.hasData) return result;

    const readNumber = (obj, keys) => {
      for (const key of keys) {
        if (obj && Object.prototype.hasOwnProperty.call(obj, key)) {
          const val = Number(obj[key]);
          if (Number.isFinite(val)) return val;
        }
      }
      return null;
    };
    const seen = new WeakSet();
    const walk = value => {
      if (!value || typeof value !== "object" || seen.has(value)) return;
      seen.add(value);
      if (!Array.isArray(value)) {
        const pnl = readNumber(value, ["realisedPNL", "realizedPNL", "realisedPnl", "realizedPnl", "realised_pnl", "realized_pnl"]);
        const fee = readNumber(value, ["tradingFee", "tradingfee", "trading_fee"]);
        if (pnl !== null || fee !== null) {
          result.hasData = true;
          result.pnl += pnl || 0;
          result.fee += fee || 0;
        }
      }
      Object.keys(value).forEach(key => walk(value[key]));
    };
    walk(historyData);
    return result;
  }

      let pnl = NaN;

        const pnlMatch = text.match(/Total\s+PNL:\s*([+-]?\d+(?:[.,]\d+)?)/i);
        if (pnlMatch) pnl = parseNumberFromText(pnlMatch[1]);

        const url = endPoint + "/getHistoryData?y=" + today.getFullYear() + "&m=" + today.getMonth() + "&d=" + today.getDate() + "&daysBefore=1";
        const controller = typeof AbortController === "function" ? new AbortController() : null;
        const timer = controller ? setTimeout(() => controller.abort(), 15000) : 0;
        const response = await fetch(url, { method: "GET", headers: { "Content-Type": "application/json", "Authorization": "Bearer " + token }, signal: controller?.signal });
        if (timer) clearTimeout(timer);
        const data = await response.json();
        if (!response.ok || data.res === "Error") throw new Error(data?.msg || "getHistoryData failed");
        let totals = sumHistoryPnlAndFee(data.answer);
        if (!totals.hasData || (totals.pnl === 0 && totals.fee === 0)) {
          totals = await loadHistoryTotalsFromPage();
        }
        if (!totals.hasData || (totals.pnl === 0 && totals.fee === 0)) throw new Error("history totals not found");
        const pnl = totals.hasData ? totals.pnl : 0;
        const fee = totals.hasData ? totals.fee : 0;
        localStorage.setItem("at_total_pnl_today", pnl);
        localStorage.setItem("at_total_fee_today", fee);
        render(pnl, fee, false, false);
      } catch (err) {
        console.error("[MIX Total PNL]", err);
        render(parseFloat(localStorage.getItem("at_total_pnl_today") || "0"), parseFloat(localStorage.getItem("at_total_fee_today") || "0"), false, true);
      }
    };
    if (btn.dataset.mixLoading !== "true") {
      render(parseFloat(localStorage.getItem("at_total_pnl_today") || "0"), parseFloat(localStorage.getItem("at_total_fee_today") || "0"), false, false);
    }

  async function fetchLeaderboardData(btn) {
    if (isLeaderboardFetching || !isOn("atLeaderboard")) return;
    isLeaderboardFetching = true;
    const original = btn.innerHTML;
    btn.innerHTML = "⏳";
    btn.style.pointerEvents = "none";
    renderLeaderboard("<span class=\"lb-empty\">Завантаження даних...</span>");
    let iframe = null;
    try {
      iframe = document.createElement("iframe");
      iframe.style.cssText = "position:absolute;left:-9999px;top:-9999px;width:1280px;height:800px;opacity:.001;pointer-events:none;z-index:-1;";
      document.body.appendChild(iframe);
      await new Promise(resolve => { iframe.onload = resolve; iframe.src = "/leaderboard"; });
      await delay(4000);

      const doc = iframe.contentDocument || iframe.contentWindow.document;
      const traders = new Map();
      doc.querySelectorAll("table").forEach(table => {
        let traderName = "Unknown";
        let card = table.parentElement;
        while (card && card !== doc.body) {
          const className = String(card.className || "");
          if (className.includes("bg-white") || className.includes("rounded") || className.includes("card") || className.includes("shadow")) break;
          card = card.parentElement;
        }
        if (card && card !== doc.body) {
          const walker = doc.createTreeWalker(card, NodeFilter.SHOW_TEXT);
          let node;
          while ((node = walker.nextNode())) {
            const match = node.nodeValue.trim().match(/^\d+\.\s*(.+)/);
            if (match) { traderName = match[1].trim(); break; }
          }
        }
        if (traderName === "Unknown") {
          let walk = table.previousElementSibling;
          while (walk) {
            const match = walk.textContent.trim().match(/^\d+\.\s*(.+)/);
            if (match) { traderName = match[1].trim(); break; }
            walk = walk.previousElementSibling;
          }
        }

        const positions = {};
        let currentExchange = "UNKNOWN";
        Array.from(table.querySelectorAll("tr")).filter(row => {
          const text = row.textContent.toLowerCase();
          return !text.includes("exchange") && !text.includes("symbol") && text.trim() !== "";
        }).forEach(row => {
          const cells = Array.from(row.querySelectorAll("td"));
          if (!cells.length) return;
          let sideIndex = -1;
          let side = "";
          cells.forEach((cell, index) => {
            const text = cell.textContent.trim().toUpperCase();
            if (text === "LONG" || text === "SHORT") {
              sideIndex = index;
              side = text === "SHORT" ? "S" : "L";
            }
          });
          if (sideIndex <= 0) return;
          const symbol = cells[sideIndex - 1]?.textContent.trim();
          if (sideIndex >= 2) {
            const exchangeText = cells[sideIndex - 2]?.textContent.trim();
            if (exchangeText) currentExchange = exchangeText;
          }
          if (!symbol) return;
          if (!positions[symbol]) positions[symbol] = [];
          positions[symbol].push({ exchange: currentExchange, side });
        });

        Object.entries(positions).forEach(([symbol, legs]) => {
          if (!legs.length) return;
          const legsHtml = legs.map(leg => {
            const color = leg.side === "S" ? "#dc3545" : "#28a745";
            return "<span style=\"color:" + color + ";font-weight:700\">" + leg.exchange + "(" + leg.side + ")</span>";
          }).join(" / ");
          if (!traders.has(traderName)) traders.set(traderName, []);
          traders.get(traderName).push("<b>" + symbol + "</b> (" + legsHtml + ")");
        });
      });

      const grouped = Array.from(traders.entries()).map(([traderName, positions]) =>
        "<span class=\"lb-data\">👤 " + traderName + ": " + positions.join(", ") + "</span>"
      );
      const htmlData = grouped.length
        ? grouped.join(" <span style=\"color:#ccc;margin:0 5px;\">|</span> ")
        : "<span class=\"lb-empty\">Немає активних позицій</span>";
      localStorage.setItem("at_leaderboard_data", htmlData);
      renderLeaderboard(htmlData);
    } catch (err) {
      console.error("[MIX Leaderboard]", err);
      const cached = localStorage.getItem("at_leaderboard_data");
      renderLeaderboard(cached || "<span class=\"lb-empty\" style=\"color:#dc3545;\">Помилка завантаження даних</span>");
    } finally {
      iframe?.remove();
      btn.innerHTML = original;
      btn.style.pointerEvents = "auto";
      isLeaderboardFetching = false;
    }
  }

  function rememberDraggedCardPlace(card) {
    const container = card?.closest?.(".trade-cards-group");
    if (!container || isCardFavorite(card)) return;
    let prev = card.previousElementSibling;
    let next = card.nextElementSibling;
    while (prev && !prev.matches?.(".trade-card.active")) prev = prev.previousElementSibling;
    while (next && !next.matches?.(".trade-card.active")) next = next.nextElementSibling;
    const places = loadManualCardPlaces();
    places[cardId(card)] = {
      prevId: prev ? cardId(prev) : "",
      nextId: next ? cardId(next) : "",
      updatedAt: Date.now()
    };
    saveManualCardPlaces(places);
  }

  function applyManualCardPlaces() {
    if (dragState.dragging || isCardInputFocused() || !isOn("atCardDrag")) return;
    const container = $(".trade-cards-group");
    if (!container) return;
    const cards = getActiveCards(container);
    if (cards.length < 2) return;
    const activeIds = new Set(cards.map(cardId));
    const places = loadManualCardPlaces();
    let changedPlaces = false;
    Object.keys(places).forEach(id => {
      if (!activeIds.has(id)) {
        delete places[id];
        changedPlaces = true;
      }
    });
    if (changedPlaces) saveManualCardPlaces(places);

    const planned = [
      ...cards.filter(isCardFavorite),
      ...cards.filter(card => !isCardFavorite(card)).sort((a, b) => cardSortKey(a).localeCompare(cardSortKey(b), undefined, { numeric: true, sensitivity: "base" }))
    ];
    const moveBefore = (card, before) => {
      const from = planned.indexOf(card);
      if (from < 0) return;
      planned.splice(from, 1);
      const to = before ? planned.indexOf(before) : -1;
      if (to >= 0) planned.splice(to, 0, card);
      else planned.push(card);
    };
    Object.entries(places)
      .sort((a, b) => Number(a[1]?.updatedAt || 0) - Number(b[1]?.updatedAt || 0))
      .forEach(([id, place]) => {
        const card = planned.find(item => cardId(item) === id);
        if (!card || isCardFavorite(card)) return;
        const next = planned.find(item => cardId(item) === place?.nextId);
        const prev = planned.find(item => cardId(item) === place?.prevId);
        if (next && next !== card) moveBefore(card, next);
        else if (prev && prev !== card) {
          const afterPrev = planned[planned.indexOf(prev) + 1] || null;
          moveBefore(card, afterPrev);
        }
      });
    const currentSignature = cards.map(cardId).join("|");
    const targetSignature = planned.map(cardId).join("|");
    if (currentSignature !== targetSignature) planned.forEach(card => container.appendChild(card));
  }

  function isFavoriteToggleActive(el) {
    const text = String(el.textContent || "");
    const cls = String(el.className || "").toLowerCase();
    const title = String(el.getAttribute("title") || "").toLowerCase();
    const label = String(el.getAttribute("aria-label") || "").toLowerCase();
    if (el.getAttribute("aria-pressed") === "true" || el.getAttribute("aria-checked") === "true") return true;
    if (/[★⭐]/.test(text) && !text.includes("☆")) return true;
    if (/\b(active|selected|checked|favorite|favorited|is-favorite|favorite-active|pinned)\b/i.test(cls)) return true;
    return /(remove|added|active|прибрати|видалити|удалить|добавлено|в обраному|в обраних|в избранном)/i.test(title + " " + label);
  }

      badge.className = "mix-at-fee-badge";
      badge.textContent = "Комісія: " + total.toFixed(2) + "%";
      badge.style.cssText = "position:absolute;top:-20px;left:0;font-size:11px;color:#fff;background:#6a6f7a;padding:2px 6px;border-radius:4px;z-index:100;";
      if (getComputedStyle(card).position === "static") card.style.position = "relative";
      card.insertBefore(badge, card.firstChild);
    });
  }

  function syncSoundButton() {
    let enabled = localStorage.getItem("at_sound_enabled") !== "false" && modules.isEnabled("sounds") !== false;
    let btn = document.getElementById("mix-at-sound-toggle-btn");
    if (!btn) {
      btn = document.createElement("button");
      btn.id = "mix-at-sound-toggle-btn";
      btn.onclick = () => { if (!isOn("atSoundButton")) modules.setEnabled("atSoundButton", true); enabled = !enabled; localStorage.setItem("at_sound_enabled", enabled); modules.setEnabled("sounds", enabled); render(); };
      getControlsContainer().appendChild(btn);
    }
    const render = () => {
      const active = isOn("atSoundButton");
      btn.style.cssText = "padding:8px 12px;background:" + (active ? (enabled ? "#28a745" : "#dc3545") : "#9ca3af") + ";color:white;border:none;border-radius:6px;font-size:12px;font-weight:600;cursor:pointer;display:flex;align-items:center;justify-content:center;height:36px;opacity:" + (active ? "1" : ".65") + ";";
      btn.textContent = enabled ? "🔊" : "🔇";
      btn.title = "Звуки MIX";
    };
    render();
  }

  function safeStep(name, fn) {
    try {
      fn();
    } catch (err) {
      console.warn("[MIX AT] module step failed:", name, err);
    }
  }

  function tickAll() {
    safeStep("installAtStyle", installAtStyle);
    safeStep("ensurePageUiVisibilityWatcher", ensurePageUiVisibilityWatcher);
    safeStep("syncPageUiVisibility", syncPageUiVisibility);
    safeStep("runFastTradeUrl", () => { void runFastTradeUrl(); });
    safeStep("runAutoSetup", runAutoSetup);
    safeStep("applyWalletsHidden", applyWalletsHidden);
    safeStep("syncHideRowsButton", syncHideRowsButton);
    safeStep("addSpinners", addSpinners);
    safeStep("syncTotalPnlButton", syncTotalPnlButton);
    safeStep("syncLeaderboardButton", syncLeaderboardButton);
    safeStep("syncBalanceSummary", syncBalanceSummary);
    safeStep("syncSignalControls", syncSignalControls);
    safeStep("applyManualCardPlaces", applyManualCardPlaces);
    safeStep("initCardDrag", initCardDrag);
    safeStep("syncFeeBadges", syncFeeBadges);
    safeStep("syncSoundButton", syncSoundButton);
    safeStep("orderToolbarButtons", orderToolbarButtons);
  }

  ["atAutoSetup", "atWalletsHidden", "atBalanceSummary", "atSpinners", "atHideRows", "atTotalPnl", "atLeaderboard", "atSignalControls", "atCardDrag", "atFeeBadges", "atSoundButton"].forEach(id => {
    modules.onToggle(id, tickAll);
  });
  document.addEventListener("mix:pnl-settings-changed", tickAll);

  tickAll();
  setInterval(tickAll, 1500);
})();
