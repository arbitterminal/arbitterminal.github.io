  function normalizeOrderUsdt(value) {
    const n = Number(value);
    if (!Number.isFinite(n)) return DEFAULT_LIVE_ORDER_USDT;
    return ORDER_SIZE_OPTIONS.reduce((best, option) => (
      Math.abs(option - n) < Math.abs(best - n) ? option : best
    ), ORDER_SIZE_OPTIONS[0]);
  }

  function getStoredOrderUsdt() {
    try {
      return normalizeOrderUsdt(localStorage.getItem(ORDER_SIZE_STORAGE_KEY) || DEFAULT_LIVE_ORDER_USDT);
    } catch (_) {
      return DEFAULT_LIVE_ORDER_USDT;
    }
  }

  function saveOrderUsdt(value) {
    const orderUsdt = normalizeOrderUsdt(value);
    try {
      localStorage.setItem(ORDER_SIZE_STORAGE_KEY, String(orderUsdt));
    } catch (_) {}
    return orderUsdt;
  }

  function getLiveOrderUsdt() {
    return normalizeOrderUsdt(liveState.orderUsdt || getStoredOrderUsdt());
  }

  function getDirectText(el) {
    return Array.from(el.childNodes || [])
      .filter((node) => node.nodeType === Node.TEXT_NODE)
      .map((node) => node.textContent || "")
      .join(" ")
      .trim();
  }

  function getFlatText(el) {
    return String(el?.textContent || "").replace(/\s+/g, " ").trim();
  }

  function isVisibleElement(el) {
    if (!(el instanceof Element)) return false;
    const rect = el.getBoundingClientRect();
    if (rect.width <= 0 || rect.height <= 0) return false;
    const style = getComputedStyle(el);
    return style.display !== "none" && style.visibility !== "hidden" && style.opacity !== "0";
  }

  function hasAncestorText(el, regex) {
    let node = el;
    for (let i = 0; node && i < 5; i += 1, node = node.parentElement) {
      if (regex.test(getFlatText(node))) return true;
    }
    return false;
  }

  function findTextStyleSource(textRegex, contextRegex) {
    const control = document.getElementById(ORDER_CONTROL_ID);
    let best = null;
    document.querySelectorAll("body *").forEach((el) => {
      if (control?.contains(el) || !isVisibleElement(el)) return;
      const ownText = getDirectText(el);
      const flatText = getFlatText(el);
      if (!textRegex.test(ownText) && !textRegex.test(flatText)) return;
      if (contextRegex && !hasAncestorText(el, contextRegex)) return;

      let score = flatText.length + el.children.length * 24;
      if (textRegex.test(ownText)) score -= 120;
      if (contextRegex?.test(flatText)) score -= 30;
      if (!best || score < best.score) best = { el, score };
    });
    return best?.el || null;
  }

  function copyTextStyle(source, target) {
    if (!source || !target) return;
    const style = getComputedStyle(source);
    [
      "font-family",
      "font-size",
      "font-weight",
      "font-style",
      "font-stretch",
      "line-height",
      "letter-spacing",
      "color",
      "text-shadow",
      "font-variant-numeric"
    ].forEach((prop) => {
      const value = style.getPropertyValue(prop);
      if (value) target.style.setProperty(prop, value, "important");
    });
  }

  function getStoredOrderPosition() {
    try {
      const parsed = JSON.parse(localStorage.getItem(ORDER_POSITION_STORAGE_KEY) || "null");
      if (!parsed) return null;
      const left = Number(parsed.left);
      const top = Number(parsed.top);
      return Number.isFinite(left) && Number.isFinite(top) ? { left, top } : null;
    } catch (_) {
      return null;
    }
  }

  function clampOrderPosition(left, top) {
    const control = document.getElementById(ORDER_CONTROL_ID);
    const width = control?.offsetWidth || ORDER_BUTTON_WIDTH;
    const height = control?.offsetHeight || ORDER_BUTTON_HEIGHT;
    const maxLeft = Math.max(ORDER_EDGE_PADDING, window.innerWidth - width - ORDER_EDGE_PADDING);
    const maxTop = Math.max(ORDER_EDGE_PADDING, window.innerHeight - height - ORDER_EDGE_PADDING);
    return {
      left: Math.min(maxLeft, Math.max(ORDER_EDGE_PADDING, Math.round(left))),
      top: Math.min(maxTop, Math.max(ORDER_EDGE_PADDING, Math.round(top)))
    };
  }

  function updateOrderPanelDirection(left, top) {
    const control = document.getElementById(ORDER_CONTROL_ID);
    if (!control) return;
    control.classList.toggle("sr-order-panel-right", left < ORDER_PANEL_WIDTH + ORDER_PANEL_GAP + ORDER_EDGE_PADDING);
    control.classList.toggle("sr-order-panel-up", top > window.innerHeight - 150);
  }

  function applyOrderPosition(left, top) {
    const control = document.getElementById(ORDER_CONTROL_ID);
    if (!control) return;
    const pos = clampOrderPosition(left, top);
    control.style.left = `${pos.left}px`;
    control.style.top = `${pos.top}px`;
    control.style.right = "auto";
    updateOrderPanelDirection(pos.left, pos.top);
  }

  function saveOrderPosition(left, top) {
    const pos = clampOrderPosition(left, top);
    try {
      localStorage.setItem(ORDER_POSITION_STORAGE_KEY, JSON.stringify(pos));
    } catch (_) {}
    applyOrderPosition(pos.left, pos.top);
  }

      if (!el || seen.has(el) || control?.contains(el)) return;
      seen.add(el);
      const rect = el.getBoundingClientRect?.();
      if (!rect || rect.width < 10 || rect.height < 10) return;
      if (rect.width > 90 || rect.height > 90) return;
      if (rect.bottom < 0 || rect.top > window.innerHeight || rect.right < 0 || rect.left > window.innerWidth) return;
      const label = getElementLabel(el);
      const directText = getDirectText(el);
      const looksLikeSettings =
        label.includes("settings") ||
        label.includes("setting") ||
        label.includes("gear") ||
        label.includes("cog") ||
        label.includes("config") ||
        label.includes("option") ||
        label.includes("налашт") ||
        label.includes("параметр") ||
        label.includes("настрой") ||
        label.includes("шестер") ||
        label.includes("lucide-settings") ||
        label.includes("fa-gear") ||
        label.includes("fa-cog") ||
        label.includes("⚙") ||
        directText.includes("⚙");
      if (!looksLikeSettings) return;
      candidates.push({ el, rect, label, directText });
    });

    candidates.sort((a, b) => {
      const score = (candidate) => {
        const { rect, label, directText } = candidate;
        const rightGap = Math.max(0, window.innerWidth - rect.right);
        let value = rightGap * 2 + Math.abs(rect.width - 42) + Math.abs(rect.height - 42);
        if (directText.includes("⚙") || label.includes("⚙")) value -= 260;
        if (rect.top > 120) value -= 180;
        if (rect.right > window.innerWidth - 160) value -= 100;
        if (rect.top < 115) value += 160;
        return value;
      };
      const aScore = score(a);
      const bScore = score(b);
      return aScore - bScore;
    });
    return candidates[0]?.el || null;
  }

  function installOrderDrag(control, toggle, panel) {
    let drag = null;
    let suppressNextClick = false;

    toggle.addEventListener("pointerdown", (event) => {
      if (event.button !== 0) return;
      const rect = control.getBoundingClientRect();
      drag = {
        pointerId: event.pointerId,
        startX: event.clientX,
        startY: event.clientY,
        startLeft: rect.left,
        startTop: rect.top,
        moved: false
      };
      toggle.setPointerCapture?.(event.pointerId);
    });

    toggle.addEventListener("pointermove", (event) => {
      if (!drag || event.pointerId !== drag.pointerId) return;
      const dx = event.clientX - drag.startX;
      const dy = event.clientY - drag.startY;
      if (!drag.moved && Math.abs(dx) + Math.abs(dy) < 4) return;
      drag.moved = true;
      suppressNextClick = true;
      panel.hidden = true;
      control.classList.add("sr-order-dragging");
      applyOrderPosition(drag.startLeft + dx, drag.startTop + dy);
      event.preventDefault();
    });

    function finishDrag(event) {
      if (!drag || event.pointerId !== drag.pointerId) return;
      toggle.releasePointerCapture?.(event.pointerId);
      control.classList.remove("sr-order-dragging");
      if (drag.moved) {
        const rect = control.getBoundingClientRect();
        saveOrderPosition(rect.left, rect.top);
        setTimeout(() => { suppressNextClick = false; }, 0);
      }
      drag = null;
    }

    toggle.addEventListener("pointerup", finishDrag);
    toggle.addEventListener("pointercancel", finishDrag);

    return () => {
      if (!suppressNextClick) return false;
      suppressNextClick = false;
      return true;
    };
  }

  function normalizeSymbol(raw) {
    const compact = String(raw || "").trim().replace(/\s+/g, "").replace(/[|:()[\]{}<>]/g, "");
    const upper = compact.toUpperCase();
    if (upper && /USDT$/.test(upper.replace(/_/g, ""))) return upper;
    const latinBase = upper.replace(/[^A-Z0-9]/g, "");
    return latinBase ? `${latinBase}USDT` : "";
  }

  function normalizeExchange(raw) {
    const key = String(raw || "")
      .replace(/[↑↓]/g, "")
      .replace(/\s+/g, "")
      .trim()
      .toUpperCase()
      .replace(/[^A-Z.]/g, "");
    return EXCHANGE_MAP[key] || "";
  }

  function normalizeAutoCoin(raw) {
    return String(raw || "").trim().toUpperCase().replace(/[^A-Z0-9]/g, "").replace(/(?:USDT|USD|USDC)$/, "");
  }

  function normalizeAutoExchange(raw) {
    const key = String(raw || "").trim().toUpperCase().replace(/[^A-Z0-9.]/g, "").replace(/^GATEIO$/, "GATE");
    if (key === "GATE.IO") return "gate";
    if (key === "GATE") return "gate";
    if (key === "BING") return "bingx";
    return key.toLowerCase();
  }

  function clearFastTradeUrl() {
    const cleanParams = new URLSearchParams(window.location.search);
    [
      "ft_auto", "fast_auto",
      "coin", "sym", "ticker",
      "ex1", "exchange1", "long",
      "ex2", "exchange2", "short",
      "side", "signalSpread", "spread", "os"
    ].forEach(key => cleanParams.delete(key));
    const query = cleanParams.toString();
    window.history.replaceState({}, document.title, window.location.pathname + (query ? "?" + query : "") + window.location.hash);
  }

  async function runFastTradeUrl() {
    const params = new URLSearchParams(window.location.search);
    if (params.get("ft_auto") !== "1" && params.get("fast_auto") !== "1") return;
    const setupKey = window.location.search;
    if (window.__spreadRadarFastTradeUrlKey === setupKey) return;
    const consumedKey = "spreadradar_fast_trade_url_consumed_" + setupKey;
    try {
      if (sessionStorage.getItem(consumedKey) === "1") return;
      sessionStorage.setItem(consumedKey, "1");
    } catch (_) {}
    window.__spreadRadarFastTradeUrlKey = setupKey;

    const coin = normalizeAutoCoin(params.get("coin") || params.get("sym") || params.get("ticker"));
    const side = String(params.get("side") || "LONG").trim().toUpperCase() === "SHORT" ? "SHORT" : "LONG";
    const exch1Name = normalizeAutoExchange(params.get("ex1") || params.get("exchange1") || params.get("long"));
    const exch2Name = normalizeAutoExchange(params.get("ex2") || params.get("exchange2") || params.get("short"));
    const signalSpread = Number(String(params.get("signalSpread") || params.get("spread") || params.get("os") || "0").replace(",", "."));

    if (!coin || !exch1Name || !exch2Name) {
      clearFastTradeUrl();
      alert("Fast Trade URL failed: coin or exchanges are missing");
      return;
    }

    const token = localStorage.getItem("token") || "";
    const endPoint = String(localStorage.getItem("endPoint") || "").replace(/\/+$/, "");
    if (!token || !endPoint) {
      clearFastTradeUrl();
      alert("Fast Trade URL failed: ArbitTerminal login session is missing");
      return;
    }

    try {
      const response = await fetch(`${endPoint}/trading/fastStart`, {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          "Authorization": "Bearer " + token
        },
        body: JSON.stringify({
          coin,
          exch1Name,
          exch2Name,
          side,
          signalSpread: Number.isFinite(signalSpread) ? signalSpread : 0
        })
      });
      const data = await response.json().catch(() => null);
      if (!response.ok || data?.res === "Error") {
        clearFastTradeUrl();
        alert(data?.msg || `Fast Trade failed: HTTP ${response.status}`);
        return;
      }
      clearFastTradeUrl();
      setTimeout(() => {
        const refresh = Array.from(document.querySelectorAll("button"))
          .find(btn => !btn.closest(".trade-card") && btn.textContent.trim().toLowerCase().includes("refresh"));
        if (refresh) refresh.click();
        else location.reload();
      }, 500);
    } catch (err) {
      console.error("SpreadRadar Fast Trade URL failed:", err);
      clearFastTradeUrl();
      alert("Fast Trade URL failed. Check connection.");
    }
  }

  function getSymbolFromCard(card) {
    const signalSymbol = card?.querySelector?.('[data-help="signal.symbol"], td.symbol, .header .symbol, .symbol')?.textContent;
    const tradeSymbol = card?.querySelector?.(".trade-details strong")?.textContent;
    const text = String(card?.innerText || card?.textContent || "");
    const fallback = (text.match(/([^\s|:()[\]{}<>]+_?USDT)/i) || [])[1] || "";
    return normalizeSymbol(signalSymbol || tradeSymbol || fallback);
  }

  function getRouteExchanges(card) {
    let long = normalizeExchange(card?.querySelector?.(".arbitrage-info .long")?.textContent || "");
    let short = normalizeExchange(card?.querySelector?.(".arbitrage-info .short")?.textContent || "");
    const text = String(card?.innerText || card?.textContent || "").replace(/\s+/g, " ").trim();

    if (!long || !short) {
      const arrowMatches = Array.from(text.matchAll(/\b(BINANCE|MEXC|GATEIO|GATE\.IO|GATE|BINGX|BING|BITGET)\b\s*([↑↓])/gi));
      for (const match of arrowMatches) {
        const exchange = normalizeExchange(match[1]);
        if (!exchange) continue;
        if (match[2] === "↑" && !long) long = exchange;
        if (match[2] === "↓" && !short) short = exchange;
      }
    }

    if (!long || !short) {
      const found = [];
      const exchangeRegex = /\b(BINANCE|MEXC|GATEIO|GATE\.IO|GATE|BINGX|BING|BITGET)\b/gi;
      for (const match of text.matchAll(exchangeRegex)) {
        const exchange = normalizeExchange(match[1]);
        if (exchange && !found.includes(exchange)) found.push(exchange);
      }
      if (found.length >= 2) {
        long ||= found[0];
        short ||= found[1];
      }
    }

    return { long, short };
  }

  function formatRouteSymbol(ticker) {
    const symbol = normalizeSymbol(ticker);
    if (!symbol) return "";
    if (/_USDT$/i.test(symbol)) return symbol;
    if (/USDT$/i.test(symbol)) return `${symbol.slice(0, -4)}_USDT`;
    return symbol;
  }

  function getRouteKey(card) {
    const ticker = formatRouteSymbol(getSymbolFromCard(card));
    const { long, short } = getRouteExchanges(card);
    return ticker && long && short ? `${ticker}:${long}:${short}` : "";
  }

  function formatPct(value) {

  function formatAge(tsMs) {
    const ts = Number(tsMs);
    if (!Number.isFinite(ts) || ts <= 0) return "";
    const ageSec = Math.max(0, Math.round((Date.now() - ts) / 1000));
    return `${ageSec}s ago`;
  }

  function formatUsdt(value) {

  function renderSpreadChart(points) {
    const values = (Array.isArray(points) ? points : [])
      .slice(-10)
      .map((point) => Number(point?.entry_spread_pct))
      .filter((value) => Number.isFinite(value));
    if (values.length < 2) return "";

    const width = 220;
    const height = 34;
    const leftPad = 5;
    const rightPad = 5;
    const topPad = 7;
    const bottomPad = 7;
    const rawMin = Math.min(...values);
    const rawMax = Math.max(...values);
    let min = Math.min(0, rawMin);
    let max = Math.max(0, rawMax);
    let range = max - min;
    if (range < 0.001) {
      min -= 0.001;
      max += 0.001;
      range = max - min;
    }
    const padding = Math.max(range * 0.12, 0.01);
    min -= padding;
    max += padding;
    range = max - min;
    const xStep = values.length > 1 ? (width - leftPad - rightPad) / (values.length - 1) : 0;
    const yFor = (value) => {
      const ratio = (value - min) / range;
      return height - bottomPad - ratio * (height - topPad - bottomPad);
    };
    const linePoints = values
      .map((value, index) => `${(leftPad + index * xStep).toFixed(1)},${yFor(value).toFixed(1)}`)
      .join(" ");
    const last = values[values.length - 1];
    const lineColor = last < 0 ? "#facc15" : "#4ade80";
    const fillColor = last < 0 ? "rgba(250,204,21,.12)" : "rgba(74,222,128,.12)";
    const baselineY = yFor(0).toFixed(1);
    const areaPoints = `${leftPad},${baselineY} ${linePoints} ${width - rightPad},${baselineY}`;

    return `
      <div class="sr-live-chart" aria-hidden="true">
        <svg viewBox="0 0 ${width} ${height}" preserveAspectRatio="none">
          <line class="sr-chart-zero" x1="${leftPad}" y1="${baselineY}" x2="${width - rightPad}" y2="${baselineY}"></line>
          <polygon class="sr-chart-fill" points="${areaPoints}" style="fill:${fillColor}"></polygon>
          <polyline class="sr-chart-line" points="${linePoints}" style="stroke:${lineColor}"></polyline>
        </svg>
      </div>
    `;
  }

  async function pollLiveSpread() {
    if (!liveState.routeKey || liveState.inFlight) return;
    liveState.inFlight = true;
    liveState.controller?.abort?.();
    liveState.controller = new AbortController();

    try {
      const orderUsdt = getLiveOrderUsdt();
      const params = new URLSearchParams({ route_key: liveState.routeKey, limit: "10" });
      if (liveState.orderQtyBase) params.set("order_qty_base", String(liveState.orderQtyBase));
      const response = await fetch(`${LIVE_SPREAD_URL}?${params}`, {
        credentials: "include",
        signal: liveState.controller.signal
      });
      const data = await response.json();
      if (!response.ok) {
        if (response.status === 402 || data?.error === "subscription_required") {
          renderLiveTip('<div class="sr-live-muted">SpreadRadar login/subscription required</div>');
          return;
        }
        throw new Error(`HTTP ${response.status}`);
      }
      const points = Array.isArray(data?.points) ? data.points : [];
      const last = points[points.length - 1];
      if (!last) {
        renderLiveTip('<div class="sr-live-muted">No live data</div>');
        return;
      }
      if (!liveState.orderQtyBase) {
        const longAsk = Number(last.long_ask);
        if (Number.isFinite(longAsk) && longAsk > 0) {
          liveState.orderQtyBase = orderUsdt / longAsk;
          liveState.inFlight = false;
          return await pollLiveSpread();
        }
      }

      const orderLabel = formatUsdt(last.order_notional_usdt || orderUsdt);
      const depthText = last.entry_order_depth_ok === false
        ? '<span class="sr-live-title">DEPTH:</span> <span class="sr-live-depth-no">NO</span>'
        : '<span class="sr-live-title">DEPTH:</span> <span class="sr-live-depth-ok">OK</span>';
      const openClass = Number(last.entry_spread_pct) < 0 ? "sr-live-warning" : "sr-live-value";
      const chartHtml = renderSpreadChart(points);
      renderLiveTip(`
        <div class="sr-live-row"><span><span class="sr-live-title">OPEN:</span> <span class="sr-live-money">${orderLabel}</span></span><span class="${openClass} sr-live-spread">${formatPct(last.entry_spread_pct)}</span></div>
        <div class="sr-live-row"><span class="sr-live-label">${depthText}</span><span class="sr-live-muted">${formatAge(last.ts_ms)}</span></div>
        ${chartHtml}
      `);
    } catch (err) {
      if (err?.name !== "AbortError") renderLiveTip('<div class="sr-live-muted">Live unavailable</div>');
    } finally {
      liveState.inFlight = false;
    }
  }

  function startLiveHover(tickerEl) {
    const card = tickerEl.closest(".signal-card");
    if (!card) return;
    const routeKey = getRouteKey(card);
    if (!routeKey) return;
    if (liveState.tickerEl === tickerEl && liveState.routeKey === routeKey) return;

    stopLiveHover();
    liveState.tickerEl = tickerEl;
    liveState.card = card;
    liveState.routeKey = routeKey;
    liveState.orderUsdt = getStoredOrderUsdt();
    renderLiveTip('<div class="sr-live-muted">Loading live spread...</div>');
    pollLiveSpread();
    liveState.timer = setInterval(pollLiveSpread, LIVE_POLL_MS);
  }

    openSpreadRadar(card);
  }

  function isUtilityPage() {
    const path = location.pathname.toLowerCase();
    return /history|trend|chart|leaderboard|cooperation|settings|login/.test(path)
      || !!document.querySelector(".chart-fullscreen-btn:fullscreen");
  }

  function hideSignalCard(card) {
    if (!card) return;
    card.dataset.spreadradarHidden = "true";
    card.style.setProperty("display", "none", "important");
    if (liveState.card === card) stopLiveHover();
  }

  function syncSignalPanel() {
    if (isUtilityPage()) {
      document.body.classList.remove("spreadradar-signals-panel-on");
      document.querySelectorAll(".spreadradar-signal-close").forEach(btn => btn.remove());
      return;
    }

    const list = document.querySelector(".signals-list");
    document.body.classList.toggle("spreadradar-signals-panel-on", !!list && window.innerWidth > 1200);
    document.querySelectorAll(".signals-list .signal-card:not([data-spreadradar-close])").forEach(card => {
      card.dataset.spreadradarClose = "true";
      if (getComputedStyle(card).position === "static") card.style.position = "relative";
      const close = document.createElement("button");
      close.type = "button";
      close.className = "spreadradar-signal-close";
      close.textContent = "×";
      close.title = "Hide signal";
      const stopCloseEvent = event => {
        event.preventDefault();
        event.stopPropagation();
        event.stopImmediatePropagation?.();
      };
      close.addEventListener("pointerdown", stopCloseEvent, true);
      close.addEventListener("mousedown", stopCloseEvent, true);
      close.addEventListener("click", event => {
        stopCloseEvent(event);
        hideSignalCard(card);
      }, true);
      card.appendChild(close);
    });
  }

  injectStyle();
  ensureOrderControl();
  runFastTradeUrl();
  syncSignalPanel();
  setInterval(syncSignalPanel, 1000);
  document.addEventListener("DOMContentLoaded", ensureOrderControl, { once: true });
  document.addEventListener("click", onClick, true);
  document.addEventListener("mouseover", onHoverIn, true);
  document.addEventListener("mouseout", onHoverOut, true);
  window.addEventListener("resize", syncSignalPanel, { passive: true });

})();
